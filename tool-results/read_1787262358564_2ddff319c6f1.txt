   158→// ==================== HELPERS ====================
   159→function formatDate(dateStr: string): string {
   160→  const d = new Date(dateStr);
   161→  return d.toLocaleDateString('ar-SA', { year: 'numeric', month: 'short', day: 'numeric' });
   162→}
   163→
   164→function formatDateShort(dateStr: string): string {
   165→  const d = new Date(dateStr);
   166→  return d.toLocaleDateString('ar-SA', { month: 'short', day: 'numeric' });
   167→}
   168→
   169→function todayISO(): string {
   170→  return new Date().toISOString().split('T')[0];
   171→}
   172→
   173→function daysAgoISO(days: number): string {
   174→  const d = new Date();
   175→  d.setDate(d.getDate() - days);
   176→  return d.toISOString().split('T')[0];
   177→}

   235→// ==================== MAIN COMPONENT ====================
   236→export function ReportsPage() {
   237→  const { t, locale } = useTranslation();
   238→
   239→  // Local translation with fallback to Arabic
   240→  const lt = useCallback((key: string): string => {
   241→    return localTranslations[locale]?.[key] || localTranslations.ar[key] || key;
   242→  }, [locale]);
   243→
   244→  // State
   245→  const [period, setPeriod] = useState<string>('monthly');
   246→  const [startDate, setStartDate] = useState<string>(daysAgoISO(30));
   247→  const [endDate, setEndDate] = useState<string>(todayISO());
   248→  const [reportData, setReportData] = useState<ReportData | null>(null);
   249→  const [loading, setLoading] = useState(true);
   250→  const [exporting, setExporting] = useState(false);
   251→
   252→  // Fetch reports data
   253→  const fetchReport = useCallback(async () => {
   254→    setLoading(true);
   255→    try {
   256→      const params = new URLSearchParams({
   257→        period,
   258→        startDate,
   259→        endDate,
   260→      });
   261→      const res = await fetch(`/api/reports?${params}`);
   262→      if (!res.ok) throw new Error('Failed');
   263→      const data = await res.json();
   264→      setReportData(data);
   265→    } catch {
   266→      toast.error(lt('reports.fetchFailed'));
   267→    } finally {
   268→      setLoading(false);
   269→    }
   270→  }, [period, startDate, endDate, lt]);
   271→
   272→  useEffect(() => {
   273→    fetchReport();
   274→  }, [fetchReport]);
   275→
   276→  // Reset to defaults
   277→  const handleReset = () => {
   278→    setPeriod('monthly');
   279→    setStartDate(daysAgoISO(30));
   280→    setEndDate(todayISO());
   281→  };
   282→
   283→  // Export as JSON
   284→  const handleExport = async () => {
   285→    setExporting(true);
   286→    try {
   287→      const params = new URLSearchParams({ period, startDate, endDate });
   288→      const res = await fetch(`/api/reports?${params}`);
   289→      if (!res.ok) throw new Error('Failed');
   290→      const data = await res.json();
   291→      const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
   292→      const url = URL.createObjectURL(blob);
   293→      const a = document.createElement('a');
   294→      a.href = url;
   295→      a.download = `wms-report-${startDate}-${endDate}.json`;
   296→      a.click();
   297→      URL.revokeObjectURL(url);
   298→      toast.success(lt('reports.exportSuccess'));
   299→    } catch {
   300→      toast.error(lt('reports.exportFailed'));
   301→    } finally {
   302→      setExporting(false);
   303→    }
   304→  };
   305→
   306→  // Calculate KPIs
   307→  const kpis = useMemo(() => {
   308→    if (!reportData) return { totalMovements: 0, avgDwell: 0, utilization: 0, critical: 0 };
   309→    const totalMovements = reportData.movementStats.total;
   310→    const avgDwell = reportData.dwellTime.avgDays;
   311→    const locUtil = Array.isArray(reportData.locationUtilization) ? reportData.locationUtilization : [];
   312→    const utilization = locUtil.length > 0
   313→      ? Math.round(locUtil.reduce((s, l) => s + l.percentage, 0) / locUtil.length)
   314→      : 0;
   315→    const critical = reportData.dwellTime.critical;
   316→    return { totalMovements, avgDwell, utilization, critical };
   317→  }, [reportData]);
   318→
   319→  // Donut chart data
   320→  const donutData = useMemo(() => {
   321→    if (!Array.isArray(reportData?.statusDistribution) || reportData.statusDistribution.length === 0) return { data: [], gradient: '' };
   322→    const entries = reportData.statusDistribution;
   323→    const total = entries.reduce((s, e) => s + e.count, 0);
   324→    let cumulative = 0;
   325→    const segments = entries.map((entry) => {
   326→      const start = (cumulative / total) * 100;
   327→      cumulative += entry.count;
   328→      const end = (cumulative / total) * 100;
   329→      const color = STATUS_COLORS[entry.status] || '#64748b';
   330→      return { ...entry, start, end, color };
   331→    });
   332→    const gradient = segments.map((s) => `${s.color} ${s.start}% ${s.end}%`).join(', ');
   333→    return { data: segments, gradient };
   334→  }, [reportData]);
   335→
   336→  // Max values for bar charts
   337→  const maxFlow = useMemo(() => {
   338→    if (!reportData?.cargoFlow) return 0;
   339→    return Math.max(1, ...reportData.cargoFlow.map((d) => Math.max(d.inbound, d.outbound)));
   340→  }, [reportData]);
   341→
   342→  const maxWeight = useMemo(() => {
   343→    if (!reportData?.weightStats?.byCategory) return 0;
   344→    return Math.max(1, ...reportData.weightStats.byCategory.map((c) => c.weight));
   345→  }, [reportData]);
   346→
   347→  const maxMovement = useMemo(() => {
   348→    if (!reportData?.movementStats?.byType) return 0;
   349→    return Math.max(1, ...reportData.movementStats.byType.map((m) => m.count));
   350→  }, [reportData]);
   351→
   352→  return (
   353→    <div className="space-y-6">
   354→      {/* ===== PAGE HEADER ===== */}
   355→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   356→        <div>
   357→          <h1 className="text-2xl font-bold dark:text-slate-100 text-slate-900">{lt('reports.title')}</h1>
   358→          <p className="text-sm dark:text-slate-500 text-slate-400 mt-1">{lt('reports.subtitle')}</p>
   359→        </div>
   360→        <Button
   361→          onClick={handleExport}
   362→          disabled={exporting || loading}
   363→          className="bg-gradient-to-l from-amber-500 to-amber-600 hover:from-amber-600 hover:to-amber-700 text-slate-900 font-medium shadow-lg shadow-amber-500/20 transition-all duration-200 hover:shadow-xl hover:shadow-amber-500/30"
   364→        >
   365→          {exporting ? <Loader2 className="h-4 w-4 ml-2 animate-spin" /> : <FileJson className="h-4 w-4 ml-2" />}
   366→          {lt('reports.export')}
   367→        </Button>
   368→      </div>
   369→
   370→      {/* ===== FILTER BAR ===== */}
   371→      <Card className="dark:border-slate-800 border-slate-200 dark:bg-slate-900/50 bg-white">
   372→        <CardContent className="p-4">
   373→          <div className="flex flex-col gap-4 sm:flex-row sm:items-end sm:flex-wrap">
   374→            {/* Period selector */}
   375→            <div className="flex flex-col gap-1.5 min-w-0 sm:w-auto">
   376→              <label className="text-xs font-medium dark:text-slate-500 text-slate-400">{lt('reports.period')}</label>
   377→              <Select value={period} onValueChange={setPeriod}>
   378→                <SelectTrigger className="w-full sm:w-[160px] dark:bg-slate-800/60 bg-slate-50 dark:border-slate-700 border-slate-300">
   379→                  <SelectValue />
   380→                </SelectTrigger>
   381→                <SelectContent>
   382→                  <SelectItem value="daily">{lt('reports.daily')}</SelectItem>
   383→                  <SelectItem value="weekly">{lt('reports.weekly')}</SelectItem>
   384→                  <SelectItem value="monthly">{lt('reports.monthly')}</SelectItem>
   385→                  <SelectItem value="yearly">{lt('reports.yearly')}</SelectItem>
   386→                </SelectContent>
   387→              </Select>
   388→            </div>
   389→
   390→            {/* Date range */}
   391→            <div className="flex flex-col gap-1.5">
   392→              <label className="text-xs font-medium dark:text-slate-500 text-slate-400">{lt('reports.from')}</label>
   393→              <input
   394→                type="date"
   395→                value={startDate}
   396→                onChange={(e) => setStartDate(e.target.value)}
   397→                className="h-9 rounded-md border dark:bg-slate-800/60 bg-slate-50 dark:border-slate-700 border-slate-300 px-3 text-sm dark:text-slate-200 text-slate-800 focus:outline-none focus:ring-2 focus:ring-amber-500/30 focus:border-amber-500 transition-colors"
   398→              />
   399→            </div>
   400→            <div className="flex flex-col gap-1.5">
   401→              <label className="text-xs font-medium dark:text-slate-500 text-slate-400">{lt('reports.to')}</label>
   402→              <input
   403→                type="date"
   404→                value={endDate}
   405→                onChange={(e) => setEndDate(e.target.value)}
   406→                className="h-9 rounded-md border dark:bg-slate-800/60 bg-slate-50 dark:border-slate-700 border-slate-300 px-3 text-sm dark:text-slate-200 text-slate-800 focus:outline-none focus:ring-2 focus:ring-amber-500/30 focus:border-amber-500 transition-colors"
   407→              />
   408→            </div>
   409→
   410→            {/* Action buttons */}
   411→            <div className="flex gap-2">
   412→              <Button
   413→                size="sm"
   414→                onClick={fetchReport}
   415→                disabled={loading}
   416→                className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium transition-all duration-200"
   417→              >
   418→                {loading ? <Loader2 className="h-3.5 w-3.5 ml-1.5 animate-spin" /> : <Filter className="h-3.5 w-3.5 ml-1.5" />}
   419→                {lt('reports.apply')}
   420→              </Button>
   421→              <Button variant="outline" size="sm" onClick={handleReset} className="dark:border-slate-700 border-slate-300 dark:text-slate-400 text-slate-600 dark:hover:bg-slate-800 hover:bg-slate-100 transition-all duration-200">
   422→                <RotateCcw className="h-3.5 w-3.5 ml-1.5" />
   423→                {lt('reports.reset')}
   424→              </Button>
   425→            </div>
   426→          </div>
   427→        </CardContent>
   428→      </Card>
   429→
   430→      {/* ===== KPI CARDS ROW ===== */}
   431→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   432→        <KPICard
   433→          icon={ArrowRightLeft}
   434→          label={lt('reports.kpi.movements')}
   435→          value={loading ? '...' : kpis.totalMovements.toLocaleString()}
   436→          loading={loading}
   437→          colorClass="dark:bg-emerald-500/10 bg-emerald-500/10 text-emerald-500 dark:text-emerald-400"
   438→          accentFrom="from-emerald-500"
   439→          accentTo="to-teal-500"
   440→          trend={loading ? undefined : `${lt('reports.total')}: ${reportData?.movementStats?.total || 0}`}
   441→        />
   442→        <KPICard
   443→          icon={Clock}
   444→          label={lt('reports.kpi.dwellTime')}
   445→          value={loading ? '...' : kpis.avgDwell}
