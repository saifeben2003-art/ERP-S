    54→
    55→type CertStatus = 'valid' | 'expiring' | 'expired' | 'none';
    56→
    57→function getCertStatus(dateStr: string | null): CertStatus {
    58→  if (!dateStr) return 'none';
    59→  const diff = (new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24);
    60→  if (diff < 0) return 'expired';
    61→  if (diff <= 30) return 'expiring';
    62→  return 'valid';
    63→}
    64→
    65→function getCertDays(dateStr: string | null): number {
    66→  if (!dateStr) return 0;
    67→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    68→}

   715→function CertStatusCard({ certStatus, certDays, certId, certExpiry, t, textPrimary, textSecondary, textTertiary }: {
   716→  certStatus: CertStatus;
   717→  certDays: number;
   718→  certId: string | null;
   719→  certExpiry: string | null;
   720→  t: (key: string) => string;
   721→  textPrimary: string;
   722→  textSecondary: string;
   723→  textTertiary: string;
   724→}) {
   725→  const config = {
   726→    valid: { bg: 'bg-emerald-500/10', border: 'border-emerald-500/20', text: 'text-emerald-400', icon: CheckCircle2, label: t('detail.equipment.certValid') },
   727→    expiring: { bg: 'bg-amber-500/10', border: 'border-amber-500/20', text: 'text-amber-400', icon: Clock, label: t('detail.equipment.certExpiring') },
   728→    expired: { bg: 'bg-red-500/10', border: 'border-red-500/20', text: 'text-red-400', icon: XCircle, label: t('detail.equipment.certExpired') },
   729→    none: { bg: 'dark:bg-slate-800 bg-slate-100', border: 'dark:border-slate-700 border-slate-200', text: textTertiary, icon: ShieldCheck, label: t('detail.equipment.certNone') },
   730→  }[certStatus];
   731→
   732→  const IconComp = config.icon;
   733→
   734→  return (
   735→    <div className="space-y-3">
   736→      {/* Main status indicator */}
   737→      <div className={`flex items-center gap-3 p-3 rounded-lg border ${config.bg} ${config.border} transition-all duration-200`}>
   738→        <div className={`w-10 h-10 rounded-full flex items-center justify-center ${config.bg} shrink-0`}>
   739→          <IconComp className={`h-5 w-5 ${config.text}`} />
   740→        </div>
   741→        <div className="flex-1 min-w-0">
   742→          <p className={`text-xs font-semibold ${config.text}`}>{config.label}</p>
   743→          {certStatus !== 'none' && certExpiry && (
   744→            <p className={`text-[10px] ${textTertiary} mt-0.5`}>
   745→              {certStatus === 'expired'
   746→                ? `${Math.abs(certDays)} ${t('detail.equipment.daysOverdue')}`
   747→                : `${certDays} ${t('detail.equipment.daysRemaining')}`}
   748→            </p>
   749→          )}
   750→        </div>
   751→      </div>
   752→      {/* Cert details */}
   753→      {certId && (
   754→        <div className="flex items-center justify-between">
   755→          <span className={`text-[10px] ${textTertiary}`}>{t('detail.equipment.certId')}</span>
   756→          <span className={`text-xs font-mono ${textPrimary}`}>{certId}</span>
   757→        </div>
   758→      )}
   759→      {certExpiry && (
   760→        <div className="flex items-center justify-between">
   761→          <span className={`text-[10px] ${textTertiary}`}>{t('detail.equipment.certExpiryDate')}</span>
   762→          <span className={`text-xs ${textPrimary}`}>{new Date(certExpiry).toLocaleDateString()}</span>
   763→        </div>
   764→      )}
   765→    </div>
   766→  );
   767→}
