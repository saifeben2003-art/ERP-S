   101→// ==================== WEIGHT CATEGORIES ====================
   102→
   103→interface WeightCategory {
   104→  label: string;
   105→  maxTonnes: number;
   106→  color: string;
   107→  colorDark: string;
   108→}
   109→
   110→const WEIGHT_CATEGORIES: WeightCategory[] = [
   111→  { label: 'Light', maxTonnes: 5, color: 'bg-emerald-400', colorDark: 'bg-emerald-500' },
   112→  { label: 'Medium', maxTonnes: 50, color: 'bg-amber-400', colorDark: 'bg-amber-500' },
   113→  { label: 'Heavy', maxTonnes: 200, color: 'bg-orange-500', colorDark: 'bg-orange-500' },
   114→  { label: 'Super Heavy', maxTonnes: 500, color: 'bg-red-500', colorDark: 'bg-red-500' },
   115→];
   116→
   117→function getWeightCategory(weightKg: number): WeightCategory {
   118→  const tonnes = weightKg / 1000;
   119→  for (let i = WEIGHT_CATEGORIES.length - 1; i >= 0; i--) {
   120→    if (tonnes > WEIGHT_CATEGORIES[i].maxTonnes) {
   121→      return { ...WEIGHT_CATEGORIES[i] };
   122→    }
   123→  }
   124→  return WEIGHT_CATEGORIES[0];
   125→}
   126→
   127→function getWeightPercentage(weightKg: number): number {
   128→  const tonnes = weightKg / 1000;
   129→  return Math.min(100, (tonnes / 500) * 100);
   130→}

   178→      ...(statusFilter && { status: statusFilter }),

   192→    try {
   193→      const [locRes, projRes] = await Promise.all([
   194→        fetch('/api/locations?limit=100'),
   195→        fetch('/api/projects?limit=100'),
   196→      ]);

   198→      const projData = await projRes.json();
   199→      setLocations(locData.items || []);
   200→      setProjects(projData.items || []);
   201→    } catch { /* silent */ }

   212→    const payload: Record<string, unknown> = {
   213→      description: form.description,
   214→      weight: parseFloat(form.weight) || 0,

   217→      height: parseFloat(form.height) || 0,
   218→      liftCategory: form.liftCategory,
   219→      commodityType: form.commodityType,
   220→      specialHandling: form.specialHandling || null,
   221→      clientName: form.clientName || null,
   222→      poReference: form.poReference || null,
   223→      blReference: form.blReference || null,
   224→      centerOfGravity: form.centerOfGravity || null,
   225→      liftingPoints: form.liftingPoints ? parseInt(form.liftingPoints) : null,
   226→      projectId: form.projectId && form.projectId !== '_none' ? form.projectId : null,
   227→    };

   231→        method: editing ? 'PUT' : 'POST',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify(payload),
   234→      });

   239→      toast.error(e instanceof Error ? e.message : t('cargo.toast.saveFailed'));

   262→      clientName: item.clientName || '',

   264→      blReference: item.blReference || '',
   265→      centerOfGravity: item.centerOfGravity || '',
