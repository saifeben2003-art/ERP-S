   154→// ==================== WEIGHT CATEGORIES ====================
   155→
   156→interface WeightCategory {
   157→  label: string;
   158→  maxTonnes: number;
   159→  color: string;
   160→  colorDark: string;
   161→}
   162→
   163→const WEIGHT_CATEGORIES: WeightCategory[] = [
   164→  { label: 'Light', maxTonnes: 5, color: 'bg-emerald-400', colorDark: 'bg-emerald-500' },
   165→  { label: 'Medium', maxTonnes: 50, color: 'bg-amber-400', colorDark: 'bg-amber-500' },
   166→  { label: 'Heavy', maxTonnes: 200, color: 'bg-orange-500', colorDark: 'bg-orange-500' },
   167→  { label: 'Super Heavy', maxTonnes: 500, color: 'bg-red-500', colorDark: 'bg-red-500' },
   168→];
   169→
   170→function getWeightCategory(weightKg: number): WeightCategory {
   171→  const tonnes = weightKg / 1000;
   172→  for (let i = WEIGHT_CATEGORIES.length - 1; i >= 0; i--) {
   173→    if (tonnes > WEIGHT_CATEGORIES[i].maxTonnes) {
   174→      return { ...WEIGHT_CATEGORIES[i] };
   175→    }
   176→  }
   177→  return WEIGHT_CATEGORIES[0];
   178→}
   179→
   180→function getWeightPercentage(weightKg: number): number {
   181→  const tonnes = weightKg / 1000;
   182→  return Math.min(100, (tonnes / 500) * 100);
   183→}
