   158→// ==================== HELPERS ====================
   159→function formatDate(dateStr: string): string {
   160→  const d = new Date(dateStr);
   161→  return d.toLocaleDateString('ar-SA', { year: 'numeric', month: 'short', day: 'numeric' });
   162→}
   163→
   164→function formatDateShort(dateStr: string): string {
   165→  const d = new Date(dateStr);
   166→  return d.toLocaleDateString('ar-SA', { month: 'short', day: 'numeric' });
   167→}
   168→
   169→function todayISO(): string {
   170→  return new Date().toISOString().split('T')[0];
   171→}
   172→
   173→function daysAgoISO(days: number): string {
   174→  const d = new Date();
   175→  d.setDate(d.getDate() - days);
   176→  return d.toISOString().split('T')[0];
   177→}

   235→// ==================== MAIN COMPONENT ====================
   236→export function ReportsPage() {
   237→  const { t, locale } = useTranslation();
   238→
   239→  // Local translation with fallback to Arabic
   240→  const lt = useCallback((key: string): string => {
   241→    return localTranslations[locale]?.[key] || localTranslations.ar[key] || key;
   242→  }, [locale]);
   243→
   244→  // State
   245→  const [period, setPeriod] = useState<string>('monthly');
   246→  const [startDate, setStartDate] = useState<string>(daysAgoISO(30));
   247→  const [endDate, setEndDate] = useState<string>(todayISO());
   248→  const [reportData, setReportData] = useState<ReportData | null>(null);
   249→  const [loading, setLoading] = useState(true);
   250→  const [exporting, setExporting] = useState(false);
   251→
   252→  // Fetch reports data
   253→  const fetchReport = useCallback(async () => {
   254→    setLoading(true);
   255→    try {
   256→      const params = new URLSearchParams({
   257→        period,
   258→        startDate,
   259→        endDate,
   260→      });
   261→      const res = await fetch(`/api/reports?${params}`);
   262→      if (!res.ok) throw new Error('Failed');
   263→      const data = await res.json();
   264→      setReportData(data);
   265→    } catch {
   266→      toast.error(lt('reports.fetchFailed'));
   267→    } finally {
   268→      setLoading(false);
   269→    }
   270→  }, [period, startDate, endDate, lt]);
   271→
   272→  useEffect(() => {
   273→    fetchReport();
   274→  }, [fetchReport]);
   275→
   276→  // Reset to defaults
   277→  const handleReset = () => {
   278→    setPeriod('monthly');
   279→    setStartDate(daysAgoISO(30));
   280→    setEndDate(todayISO());
   281→  };
   282→
   283→  // Export as JSON
   284→  const handleExport = async () => {
   285→    setExporting(true);
   286→    try {
   287→      const params = new URLSearchParams({ period, startDate, endDate });
   288→      const res = await fetch(`/api/reports?${params}`);
   289→      if (!res.ok) throw new Error('Failed');
   290→      const data = await res.json();
   291→      const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
   292→      const url = URL.createObjectURL(blob);
   293→      const a = document.createElement('a');
   294→      a.href = url;
   295→      a.download = `wms-report-${startDate}-${endDate}.json`;
   296→      a.click();
   297→      URL.revokeObjectURL(url);
   298→      toast.success(lt('reports.exportSuccess'));
   299→    } catch {
   300→      toast.error(lt('reports.exportFailed'));
   301→    } finally {
   302→      setExporting(false);
   303→    }
   304→  };
   305→
   306→  // Calculate KPIs
   307→  const kpis = useMemo(() => {
   308→    if (!reportData) return { totalMovements: 0, avgDwell: 0, utilization: 0, critical: 0 };
   309→    const totalMovements = reportData.movementStats.total;
   310→    const avgDwell = reportData.dwellTime.avgDays;
   311→    const utilization = reportData.locationUtilization.length > 0
   312→      ? Math.round(reportData.locationUtilization.reduce((s, l) => s + l.percentage, 0) / reportData.locationUtilization.length)
   313→      : 0;
   314→    const critical = reportData.dwellTime.critical;
   315→    return { totalMovements, avgDwell, utilization, critical };
   316→  }, [reportData]);
   317→
   318→  // Donut chart data
   319→  const donutData = useMemo(() => {
   320→    if (!reportData?.statusDistribution || reportData.statusDistribution.length === 0) return { data: [], gradient: '' };
   321→    const entries = reportData.statusDistribution;
   322→    const total = entries.reduce((s, e) => s + e.count, 0);
   323→    let cumulative = 0;
   324→    const segments = entries.map((entry) => {
   325→      const start = (cumulative / total) * 100;
   326→      cumulative += entry.count;
   327→      const end = (cumulative / total) * 100;
   328→      const color = STATUS_COLORS[entry.status] || '#64748b';
   329→      return { ...entry, start, end, color };
   330→    });
   331→    const gradient = segments.map((s) => `${s.color} ${s.start}% ${s.end}%`).join(', ');
   332→    return { data: segments, gradient };
   333→  }, [reportData]);
   334→
   335→  // Max values for bar charts
   336→  const maxFlow = useMemo(() => {
   337→    if (!reportData?.cargoFlow) return 0;
   338→    return Math.max(1, ...reportData.cargoFlow.map((d) => Math.max(d.inbound, d.outbound)));
   339→  }, [reportData]);
   340→
   341→  const maxWeight = useMemo(() => {
   342→    if (!reportData?.weightStats?.byCategory) return 0;
   343→    return Math.max(1, ...reportData.weightStats.byCategory.map((c) => c.weight));
   344→  }, [reportData]);
   345→
   346→  const maxMovement = useMemo(() => {
   347→    if (!reportData?.movementStats?.byType) return 0;
   348→    return Math.max(1, ...reportData.movementStats.byType.map((m) => m.count));
   349→  }, [reportData]);
   350→
   351→  return (
   352→    <div className="space-y-6">
   353→      {/* ===== PAGE HEADER ===== */}
   354→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   355→        <div>
   356→          <h1 className="text-2xl font-bold dark:text-slate-100 text-slate-900">{lt('reports.title')}</h1>
   357→          <p className="text-sm dark:text-slate-500 text-slate-400 mt-1">{lt('reports.subtitle')}</p>
   358→        </div>
   359→        <Button
   360→          onClick={handleExport}
   361→          disabled={exporting || loading}
   362→          className="bg-gradient-to-l from-amber-500 to-amber-600 hover:from-amber-600 hover:to-amber-700 text-slate-900 font-medium shadow-lg shadow-amber-500/20 transition-all duration-200 hover:shadow-xl hover:shadow-amber-500/30"
   363→        >
   364→          {exporting ? <Loader2 className="h-4 w-4 ml-2 animate-spin" /> : <FileJson className="h-4 w-4 ml-2" />}
   365→          {lt('reports.export')}
   366→        </Button>
   367→      </div>
   368→
   369→      {/* ===== FILTER BAR ===== */}
   370→      <Card className="dark:border-slate-800 border-slate-200 dark:bg-slate-900/50 bg-white">
   371→        <CardContent className="p-4">
   372→          <div className="flex flex-col gap-4 sm:flex-row sm:items-end sm:flex-wrap">
   373→            {/* Period selector */}
   374→            <div className="flex flex-col gap-1.5 min-w-0 sm:w-auto">
   375→              <label className="text-xs font-medium dark:text-slate-500 text-slate-400">{lt('reports.period')}</label>
   376→              <Select value={period} onValueChange={setPeriod}>
   377→                <SelectTrigger className="w-full sm:w-[160px] dark:bg-slate-800/60 bg-slate-50 dark:border-slate-700 border-slate-300">
   378→                  <SelectValue />
   379→                </SelectTrigger>
   380→                <SelectContent>
   381→                  <SelectItem value="daily">{lt('reports.daily')}</SelectItem>
   382→                  <SelectItem value="weekly">{lt('reports.weekly')}</SelectItem>
   383→                  <SelectItem value="monthly">{lt('reports.monthly')}</SelectItem>
   384→                  <SelectItem value="yearly">{lt('reports.yearly')}</SelectItem>
   385→                </SelectContent>
   386→              </Select>
   387→            </div>
   388→
   389→            {/* Date range */}
   390→            <div className="flex flex-col gap-1.5">
   391→              <label className="text-xs font-medium dark:text-slate-500 text-slate-400">{lt('reports.from')}</label>
   392→              <input
   393→                type="date"
   394→                value={startDate}
   395→                onChange={(e) => setStartDate(e.target.value)}
   396→                className="h-9 rounded-md border dark:bg-slate-800/60 bg-slate-50 dark:border-slate-700 border-slate-300 px-3 text-sm dark:text-slate-200 text-slate-800 focus:outline-none focus:ring-2 focus:ring-amber-500/30 focus:border-amber-500 transition-colors"
   397→              />
   398→            </div>
   399→            <div className="flex flex-col gap-1.5">
   400→              <label className="text-xs font-medium dark:text-slate-500 text-slate-400">{lt('reports.to')}</label>
   401→              <input
   402→                type="date"
   403→                value={endDate}
   404→                onChange={(e) => setEndDate(e.target.value)}
   405→                className="h-9 rounded-md border dark:bg-slate-800/60 bg-slate-50 dark:border-slate-700 border-slate-300 px-3 text-sm dark:text-slate-200 text-slate-800 focus:outline-none focus:ring-2 focus:ring-amber-500/30 focus:border-amber-500 transition-colors"
   406→              />
   407→            </div>
   408→
   409→            {/* Action buttons */}
   410→            <div className="flex gap-2">
   411→              <Button
   412→                size="sm"
   413→                onClick={fetchReport}
   414→                disabled={loading}
   415→                className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium transition-all duration-200"
   416→              >
   417→                {loading ? <Loader2 className="h-3.5 w-3.5 ml-1.5 animate-spin" /> : <Filter className="h-3.5 w-3.5 ml-1.5" />}
   418→                {lt('reports.apply')}
   419→              </Button>
   420→              <Button variant="outline" size="sm" onClick={handleReset} className="dark:border-slate-700 border-slate-300 dark:text-slate-400 text-slate-600 dark:hover:bg-slate-800 hover:bg-slate-100 transition-all duration-200">
   421→                <RotateCcw className="h-3.5 w-3.5 ml-1.5" />
   422→                {lt('reports.reset')}
   423→              </Button>
   424→            </div>
   425→          </div>
   426→        </CardContent>
   427→      </Card>
   428→
   429→      {/* ===== KPI CARDS ROW ===== */}
   430→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   431→        <KPICard
   432→          icon={ArrowRightLeft}
   433→          label={lt('reports.kpi.movements')}
   434→          value={loading ? '...' : kpis.totalMovements.toLocaleString()}
   435→          loading={loading}
   436→          colorClass="dark:bg-emerald-500/10 bg-emerald-500/10 text-emerald-500 dark:text-emerald-400"
   437→          accentFrom="from-emerald-500"
   438→          accentTo="to-teal-500"
   439→          trend={loading ? undefined : `${lt('reports.total')}: ${reportData?.movementStats?.total || 0}`}
   440→        />
   441→        <KPICard
   442→          icon={Clock}
   443→          label={lt('reports.kpi.dwellTime')}
   444→          value={loading ? '...' : kpis.avgDwell}
   445→          suffix={lt('reports.days')}
