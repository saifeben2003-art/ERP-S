   104→// ==================== WEIGHT CATEGORIES ====================
   105→
   106→interface WeightCategory {
   107→  label: string;
   108→  maxTonnes: number;
   109→  color: string;
   110→  colorDark: string;
   111→}
   112→
   113→const WEIGHT_CATEGORIES: WeightCategory[] = [
   114→  { label: 'Light', maxTonnes: 5, color: 'bg-emerald-400', colorDark: 'bg-emerald-500' },
   115→  { label: 'Medium', maxTonnes: 50, color: 'bg-amber-400', colorDark: 'bg-amber-500' },
   116→  { label: 'Heavy', maxTonnes: 200, color: 'bg-orange-500', colorDark: 'bg-orange-500' },
   117→  { label: 'Super Heavy', maxTonnes: 500, color: 'bg-red-500', colorDark: 'bg-red-500' },
   118→];
   119→
   120→function getWeightCategory(weightKg: number): WeightCategory {
   121→  const tonnes = weightKg / 1000;
   122→  for (let i = WEIGHT_CATEGORIES.length - 1; i >= 0; i--) {
   123→    if (tonnes > WEIGHT_CATEGORIES[i].maxTonnes) {
   124→      return { ...WEIGHT_CATEGORIES[i] };
   125→    }
   126→  }
   127→  return WEIGHT_CATEGORIES[0];
   128→}
   129→
   130→function getWeightPercentage(weightKg: number): number {
   131→  const tonnes = weightKg / 1000;
   132→  return Math.min(100, (tonnes / 500) * 100);
   133→}
