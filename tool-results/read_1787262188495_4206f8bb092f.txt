   112→// ==================== COMPONENT ====================
   113→
   114→export function ScannerPage() {
   115→  const { t, locale } = useTranslation();
   116→
   117→  // Local scanner translations
   118→  const st = useCallback((key: string): string => {
   119→    const map = locale === 'en' ? scannerLocalEn : scannerLocal;
   120→    return map[key] || key;
   121→  }, [locale]);
   122→
   123→  // State
   124→  const [mode, setMode] = useState<'camera' | 'manual'>('camera');
   125→  const [manualBarcode, setManualBarcode] = useState('');
   126→  const [isScanning, setIsScanning] = useState(false);
   127→  const [scanResult, setScanResult] = useState<ScannerResponse | null>(null);
   128→  const [scanError, setScanError] = useState<string | null>(null);
   129→  const [recentScans, setRecentScans] = useState<ScanRecord[]>([]);
   130→  const [cameraActive, setCameraActive] = useState(false);
   131→  const [torchOn, setTorchOn] = useState(false);
   132→  const [facingMode, setFacingMode] = useState<'environment' | 'user'>('environment');
   133→  const [barcodeDetectorAvailable, setBarcodeDetectorAvailable] = useState(true);
   134→  const [scanLineY, setScanLineY] = useState(0);
   135→
   136→  // Refs
   137→  const videoRef = useRef<HTMLVideoElement>(null);
   138→  const canvasRef = useRef<HTMLCanvasElement>(null);
   139→  const animationRef = useRef<number>(0);
   140→  const streamRef = useRef<MediaStream | null>(null);
   141→  const scanIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   142→  const scanLineDirection = useRef<1 | -1>(1);
   143→
   144→  // ==================== BARCODE LOOKUP ====================
   145→
   146→  const lookupBarcode = useCallback(async (barcode: string) => {
   147→    if (!barcode.trim()) return;
   148→
   149→    setIsScanning(true);
   150→    setScanError(null);
   151→    setScanResult(null);
   152→
   153→    try {
   154→      const res = await fetch(`/api/scanner?barcode=${encodeURIComponent(barcode.trim())}`);
   155→      const data = await res.json();
   156→
   157→      if (res.ok) {
   158→        setScanResult(data as ScannerResponse);
   159→
   160→        // Add to recent scans
   161→        setRecentScans((prev) => {
   162→          const record: ScanRecord = {
   163→            barcode: barcode.trim(),
   164→            type: (data as ScannerResponse).type,
   165→            timestamp: Date.now(),
   166→            result: data as ScannerResponse,
   167→          };
   168→          const filtered = prev.filter((r) => r.barcode !== barcode.trim());
   169→          return [record, ...filtered].slice(0, 10);
   170→        });
   171→      } else {
   172→        setScanError(data.error || st('scanner.notFound'));
   173→        setRecentScans((prev) => {
   174→          const record: ScanRecord = {
   175→            barcode: barcode.trim(),
   176→            type: 'NOT_FOUND',
   177→            timestamp: Date.now(),
   178→            result: null,
   179→          };
   180→          const filtered = prev.filter((r) => r.barcode !== barcode.trim());
   181→          return [record, ...filtered].slice(0, 10);
   182→        });
   183→      }
   184→    } catch {
   185→      setScanError(st('scanner.notFound'));
   186→    } finally {
   187→      setIsScanning(false);
   188→    }
   189→  }, [st]);
   190→
   191→  // ==================== BARCODE DETECTION ====================
   192→
   193→  const detectBarcode = useCallback(async () => {
   194→    if (!videoRef.current || !barcodeDetectorAvailable) return;
   195→
   196→    try {
   197→      // Type assertion for BarcodeDetector which may not be in TypeScript's lib
   198→      const BarcodeDetectorAPI = (window as unknown as { BarcodeDetector?: new (opts: { formats: string[] }) => { detect: (source: HTMLVideoElement) => Promise<{ rawValue: string }[]> } }).BarcodeDetector;
   199→
   200→      if (!BarcodeDetectorAPI) return;
   201→
   202→      const detector = new BarcodeDetectorAPI({ formats: ['qr_code', 'ean_13', 'ean_8', 'code_128', 'code_39', 'upc_a', 'upc_e'] });
   203→      const barcodes = await detector.detect(videoRef.current);
   204→
   205→      if (barcodes.length > 0) {
   206→        const code = barcodes[0].rawValue;
   207→        if (code && !isScanning) {
   208→          await lookupBarcode(code);
   209→        }
   210→      }
   211→    } catch {
   212→      // Silently fail - user can use manual entry
   213→    }
   214→  }, [barcodeDetectorAvailable, isScanning, lookupBarcode]);
   215→
   216→  // ==================== CAMERA MANAGEMENT ====================
   217→
   218→  const startCamera = useCallback(async () => {
   219→    try {
   220→      // Stop existing stream
   221→      if (streamRef.current) {
   222→        streamRef.current.getTracks().forEach((track) => track.stop());
   223→      }
   224→
   225→      const constraints: MediaStreamConstraints = {
   226→        video: {
   227→          facingMode,
   228→          width: { ideal: 1280 },
   229→          height: { ideal: 720 },
   230→        },
   231→        audio: false,
   232→      };
   233→
   234→      const stream = await navigator.mediaDevices.getUserMedia(constraints);
   235→      streamRef.current = stream;
   236→
   237→      if (videoRef.current) {
   238→        videoRef.current.srcObject = stream;
   239→        await videoRef.current.play();
   240→      }
   241→
   242→      setCameraActive(true);
   243→
   244→      // Check BarcodeDetector support
   245→      const hasDetector = 'BarcodeDetector' in window;
   246→      setBarcodeDetectorAvailable(hasDetector);
   247→
   248→      if (hasDetector) {
   249→        // Scan every 500ms
   250→        scanIntervalRef.current = setInterval(detectBarcode, 500);
   251→      }
   252→    } catch {
   253→      setCameraActive(false);
   254→      setScanError(st('scanner.cameraError'));
   255→    }
   256→  }, [facingMode, detectBarcode, st]);
   257→
   258→  const stopCamera = useCallback(() => {
   259→    if (streamRef.current) {
   260→      streamRef.current.getTracks().forEach((track) => track.stop());
   261→      streamRef.current = null;
   262→    }
   263→    if (scanIntervalRef.current) {
   264→      clearInterval(scanIntervalRef.current);
   265→      scanIntervalRef.current = null;
   266→    }
   267→    if (animationRef.current) {
   268→      cancelAnimationFrame(animationRef.current);
   269→      animationRef.current = 0;
   270→    }
   271→    setCameraActive(false);
   272→  }, []);
   273→
   274→  const toggleTorch = useCallback(async () => {
   275→    if (!streamRef.current) return;
   276→    const track = streamRef.current.getVideoTracks()[0];
   277→    if (!track) return;
   278→
   279→    const capabilities = track.getCapabilities?.() as { torch?: boolean } | undefined;
   280→    if (!capabilities?.torch) return;
   281→
   282→    try {
   283→      const newTorch = !torchOn;
   284→      await track.applyConstraints({
   285→        advanced: [{ torch: newTorch } as MediaTrackConstraintSet],
   286→      });
   287→      setTorchOn(newTorch);
   288→    } catch {
   289→      // Torch not supported
   290→    }
   291→  }, [torchOn]);
   292→
   293→  const switchCamera = useCallback(() => {
   294→    setFacingMode((prev) => (prev === 'environment' ? 'user' : 'environment'));
   295→    setCameraActive(false);
   296→    if (streamRef.current) {
   297→      streamRef.current.getTracks().forEach((track) => track.stop());
   298→      streamRef.current = null;
   299→    }
   300→  }, []);
   301→
   302→  // ==================== SCAN LINE ANIMATION ====================
   303→
   304→  useEffect(() => {
   305→    if (!cameraActive || !canvasRef.current) return;
   306→
   307→    const canvas = canvasRef.current;
   308→    const ctx = canvas.getContext('2d');
   309→    if (!ctx) return;
   310→
   311→    let linePos = 0;
   312→
   313→    const animate = () => {
   314→      const w = canvas.width;
   315→      const h = canvas.height;
   316→
   317→      ctx.clearRect(0, 0, w, h);
   318→
   319→      // Draw corner brackets (viewfinder)
   320→      const bracketLen = Math.min(w, h) * 0.15;
   321→      const bracketWidth = 3;
   322→      const margin = 30;
   323→      const cornerColor = '#34d399'; // emerald-400
   324→
   325→      ctx.strokeStyle = cornerColor;
   326→      ctx.lineWidth = bracketWidth;
   327→      ctx.lineCap = 'round';
   328→
   329→      // Top-left
   330→      ctx.beginPath();
   331→      ctx.moveTo(margin, margin + bracketLen);
   332→      ctx.lineTo(margin, margin);
   333→      ctx.lineTo(margin + bracketLen, margin);
   334→      ctx.stroke();
   335→
   336→      // Top-right
   337→      ctx.beginPath();
   338→      ctx.moveTo(w - margin - bracketLen, margin);
   339→      ctx.lineTo(w - margin, margin);
   340→      ctx.lineTo(w - margin, margin + bracketLen);
   341→      ctx.stroke();
   342→
   343→      // Bottom-left
   344→      ctx.beginPath();
   345→      ctx.moveTo(margin, h - margin - bracketLen);
   346→      ctx.lineTo(margin, h - margin);
   347→      ctx.lineTo(margin + bracketLen, h - margin);
   348→      ctx.stroke();
   349→
   350→      // Bottom-right
   351→      ctx.beginPath();
   352→      ctx.moveTo(w - margin - bracketLen, h - margin);
   353→      ctx.lineTo(w - margin, h - margin);
   354→      ctx.lineTo(w - margin, h - margin - bracketLen);
   355→      ctx.stroke();
   356→
   357→      // Draw scanning line (amber gradient)
   358→      const lineY = margin + (linePos * (h - 2 * margin));
   359→      const gradient = ctx.createLinearGradient(margin, lineY, w - margin, lineY);
   360→      gradient.addColorStop(0, 'rgba(251, 191, 36, 0)');
   361→      gradient.addColorStop(0.2, 'rgba(251, 191, 36, 0.9)');
   362→      gradient.addColorStop(0.5, 'rgba(251, 146, 60, 1)');
   363→      gradient.addColorStop(0.8, 'rgba(251, 191, 36, 0.9)');
   364→      gradient.addColorStop(1, 'rgba(251, 191, 36, 0)');
   365→
   366→      ctx.strokeStyle = gradient;
   367→      ctx.lineWidth = 2;
   368→      ctx.beginPath();
   369→      ctx.moveTo(margin, lineY);
   370→      ctx.lineTo(w - margin, lineY);
   371→      ctx.stroke();
   372→
   373→      // Glow effect
   374→      const glowGradient = ctx.createLinearGradient(margin, lineY - 15, margin, lineY + 15);
   375→      glowGradient.addColorStop(0, 'rgba(251, 191, 36, 0)');
   376→      glowGradient.addColorStop(0.5, 'rgba(251, 191, 36, 0.15)');
   377→      glowGradient.addColorStop(1, 'rgba(251, 191, 36, 0)');
   378→      ctx.fillStyle = glowGradient;
   379→      ctx.fillRect(margin, lineY - 15, w - 2 * margin, 30);
   380→
   381→      // Update scan line position
   382→      linePos += 0.005 * scanLineDirection.current;
   383→      if (linePos >= 1) {
   384→        linePos = 1;
   385→        scanLineDirection.current = -1;
   386→      } else if (linePos <= 0) {
   387→        linePos = 0;
   388→        scanLineDirection.current = 1;
   389→      }
   390→
   391→      animationRef.current = requestAnimationFrame(animate);
   392→    };
   393→
   394→    animationRef.current = requestAnimationFrame(animate);
   395→
   396→    return () => {
   397→      if (animationRef.current) {
   398→        cancelAnimationFrame(animationRef.current);
   399→      }
   400→    };
   401→  }, [cameraActive]);
   402→
   403→  // ==================== CAMERA EFFECTS ====================
   404→
   405→  useEffect(() => {
   406→    if (mode === 'camera') {
   407→      startCamera();
   408→    } else {
   409→      stopCamera();
   410→    }
   411→
   412→    return () => {
   413→      stopCamera();
   414→    };
   415→  }, [mode, facingMode, startCamera, stopCamera]);
   416→
   417→  // ==================== MANUAL ENTRY ====================
   418→
   419→  const handleManualSearch = useCallback(() => {
   420→    if (manualBarcode.trim()) {
   421→      lookupBarcode(manualBarcode);
   422→    }
   423→  }, [manualBarcode, lookupBarcode]);
   424→
   425→  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
   426→    if (e.key === 'Enter') {
   427→      handleManualSearch();
   428→    }
   429→  }, [handleManualSearch]);
   430→
   431→  const handlePaste = useCallback(async () => {
   432→    try {
   433→      const text = await navigator.clipboard.readText();
   434→      if (text) {
   435→        setManualBarcode(text.trim());
   436→      }
   437→    } catch {
   438→      // Clipboard API not available
   439→    }
   440→  }, []);
   441→
   442→  const handleRecentClick = useCallback((barcode: string) => {
   443→    setManualBarcode(barcode);
   444→    setMode('manual');
   445→    lookupBarcode(barcode);
   446→  }, [lookupBarcode]);
   447→
   448→  const clearHistory = useCallback(() => {
   449→    setRecentScans([]);
   450→  }, []);
   451→
   452→  // ==================== RENDER HELPERS ====================
   453→
   454→  const formatTime = (timestamp: number) => {
   455→    const d = new Date(timestamp);
   456→    return d.toLocaleTimeString(locale === 'en' ? 'en-US' : 'ar-SA', {
   457→      hour: '2-digit',
   458→      minute: '2-digit',
   459→    });
   460→  };
   461→
   462→  const getStatusStyle = (status: string) => {
   463→    return statusStyles[status] || 'bg-slate-500/10 text-slate-500 border-slate-500/20';
   464→  };
   465→
   466→  // ==================== RENDER ====================
   467→
   468→  return (
   469→    <div className="w-full max-w-2xl mx-auto space-y-6" dir={locale === 'en' ? 'ltr' : 'rtl'}>
   470→      {/* Header */}
   471→      <div className="flex items-center gap-3">
   472→        <div className="p-2.5 rounded-xl bg-amber-500/10">
   473→          <ScanLine className="w-6 h-6 text-amber-500" />
   474→        </div>
   475→        <div>
   476→          <h1 className="text-xl font-bold text-slate-900 dark:text-white">
   477→            {st('scanner.title')}
   478→          </h1>
   479→          <p className="text-sm text-slate-500 dark:text-slate-400">
   480→            {st('scanner.subtitle')}
   481→          </p>
   482→        </div>
   483→      </div>
   484→
   485→      {/* Mode Toggle */}
   486→      <div className="flex gap-2 p-1 bg-slate-100 dark:bg-slate-800/60 rounded-xl">
   487→        <button
   488→          onClick={() => setMode('camera')}
   489→          className={`flex-1 flex items-center justify-center gap-2 py-2.5 px-4 rounded-lg text-sm font-medium transition-all ${
   490→            mode === 'camera'
   491→              ? 'bg-white dark:bg-slate-700 text-amber-600 dark:text-amber-400 shadow-sm'
   492→              : 'text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-300'
   493→          }`}
   494→        >
   495→          {cameraActive ? (
   496→            <Camera className="w-4 h-4" />
   497→          ) : (
   498→            <CameraOff className="w-4 h-4" />
   499→          )}
   500→          {st('scanner.scan')}
   501→        </button>
   502→        <button
   503→          onClick={() => setMode('manual')}
   504→          className={`flex-1 flex items-center justify-center gap-2 py-2.5 px-4 rounded-lg text-sm font-medium transition-all ${
   505→            mode === 'manual'
   506→              ? 'bg-white dark:bg-slate-700 text-amber-600 dark:text-amber-400 shadow-sm'
   507→              : 'text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-300'
   508→          }`}
   509→        >
   510→          <Type className="w-4 h-4" />
   511→          {st('scanner.manual')}
   512→        </button>
   513→      </div>
   514→
   515→      {/* Camera Scanner Area */}
   516→      {mode === 'camera' && (
   517→        <Card className="overflow-hidden border-0 dark:bg-slate-800/40 bg-slate-50">
   518→          <CardContent className="p-0">
   519→            <div className="relative aspect-[4/3] bg-slate-950 dark:bg-slate-900 rounded-t-xl overflow-hidden">
   520→              {/* Video Feed */}
   521→              <video
   522→                ref={videoRef}
   523→                className="w-full h-full object-cover"
   524→                playsInline
   525→                muted
   526→                autoPlay
   527→              />
   528→
   529→              {/* Canvas Overlay */}
   530→              <canvas
   531→                ref={canvasRef}
   532→                width={640}
   533→                height={480}
   534→                className="absolute inset-0 w-full h-full pointer-events-none"
   535→              />
   536→
   537→              {/* Camera Controls Overlay */}
   538→              <div className="absolute bottom-0 inset-x-0 bg-gradient-to-t from-black/60 to-transparent p-4">
   539→                <div className="flex items-center justify-center gap-3">
   540→                  <Button
   541→                    size="sm"
   542→                    variant="ghost"
   543→                    className="text-white/90 hover:text-white hover:bg-white/20"
