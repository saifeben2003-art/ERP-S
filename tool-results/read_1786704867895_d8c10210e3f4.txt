    70→// ==================== RELATIVE TIME ====================
    71→
    72→function relativeTime(dateStr: string): string {
    73→  const now = Date.now();
    74→  const then = new Date(dateStr).getTime();
    75→  const diffMs = now - then;
    76→  const diffMin = Math.floor(diffMs / 60000);
    77→  const diffHr = Math.floor(diffMs / 3600000);
    78→  const diffDay = Math.floor(diffMs / 86400000);
    79→  if (diffMin < 1) return 'just now';
    80→  if (diffMin < 60) return `${diffMin}m ago`;
    81→  if (diffHr < 24) return `${diffHr}h ago`;
    82→  if (diffDay < 7) return `${diffDay}d ago`;
    83→  return new Date(dateStr).toLocaleDateString();
    84→}
