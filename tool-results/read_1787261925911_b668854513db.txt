    99→export function ProjectsPage() {
   100→  const [projects, setProjects] = useState<Project[]>([]);
   101→  const [loading, setLoading] = useState(true);
   102→  const [statusFilter, setStatusFilter] = useState('');
   103→  const [showAdd, setShowAdd] = useState(false);
   104→  const [selectedProject, setSelectedProject] = useState<Project | null>(null);
   105→  const [projectCargo, setProjectCargo] = useState<CargoItem[]>([]);
   106→  const [projectMovements, setProjectMovements] = useState<Movement[]>([]);
   107→  const [detailLoading, setDetailLoading] = useState(false);
   108→  const [statusChanging, setStatusChanging] = useState(false);
   109→  const [form, setForm] = useState(emptyForm);
   110→  const [submitting, setSubmitting] = useState(false);
   111→  const { t } = useTranslation();
   112→
   113→  const fetchProjects = useCallback(async () => {
   114→    setLoading(true);
   115→    const params = new URLSearchParams({ limit: '100', ...(statusFilter && { status: statusFilter }) });
   116→    try {
   117→      const res = await fetch(`/api/projects?${params}`);
   118→      const data: ProjectListResponse = await res.json();
   119→      if (!res.ok) { setProjects([]); return; }
   120→      setProjects(data.items || []);
   121→    } catch { toast.error(t('projects.toast.fetchFailed')); } finally { setLoading(false); }
   122→  }, [statusFilter, t]);
   123→
   124→  useEffect(() => { fetchProjects(); }, [fetchProjects]);
   125→
   126→  const handleCreate = async () => {
   127→    setSubmitting(true);
   128→    try {
   129→      const res = await fetch('/api/projects', {
   130→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   131→        body: JSON.stringify({ name: form.name, description: form.description || null, clientName: form.clientName, clientContact: form.clientContact || null, destination: form.destination || null, shippingLine: form.shippingLine || null, vesselName: form.vesselName || null, etd: form.etd || null, eta: form.eta || null }),
   132→      });
   133→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   134→      toast.success(t('projects.toast.created')); setShowAdd(false); setForm(emptyForm); fetchProjects();
   135→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : t('projects.toast.createFailed')); } finally { setSubmitting(false); }
   136→  };

   138→  const openProjectDetail = async (project: Project) => {
   139→    setSelectedProject(project);
   140→    setDetailLoading(true);
   141→    try {
   142→      const [cargoRes, moveRes] = await Promise.all([
   143→        fetch(`/api/cargo?projectId=${project.id}&limit=200`),
   144→        fetch(`/api/movements?limit=100`),
   145→      ]);
   146→      const cargoData = await cargoRes.json();
   147→      const moveData = await moveRes.json();
   148→      const allMoves: Movement[] = moveData.items || [];
   149→      const cargoIds = new Set((cargoData.items || []).map((c: CargoItem) => c.id));
   150→      setProjectCargo(cargoData.items || []);
   151→      setProjectMovements(allMoves.filter((m) => cargoIds.has(m.cargoItemId)));
   152→    } catch { toast.error(t('projects.toast.cargoFetchFailed')); } finally { setDetailLoading(false); }
   153→  };

   155→  const handleProjectStatusChange = async (newStatus: ProjectStatus) => {
   156→    if (!selectedProject) return;
   157→    setStatusChanging(true);
   158→    try {
   159→      const res = await fetch(`/api/projects/${selectedProject.id}/status`, {
   160→        method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ status: newStatus }),
   161→      });
   162→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   163→      toast.success(t('detail.project.statusChanged'));
   164→      const data = await res.json();
   165→      setSelectedProject(data.data);
   166→      fetchProjects();
   167→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : t('detail.statusChangeFailed')); } finally { setStatusChanging(false); }
   168→  };
   169→
   170→  const currentStep = selectedProject ? PROJECT_STATUSES.indexOf(selectedProject.status) : -1;
   171→

   173→    (Array.isArray(projectCargo) ? projectCargo : []).reduce((acc, c) => { acc[c.status] = (acc[c.status] || 0) + 1; return acc; }, {} as Record<string, number>),
   174→    [projectCargo]
   175→  );
   176→
   177→  const completionPct = useMemo(() => {
   178→    if (!Array.isArray(projectCargo) || projectCargo.length === 0) return 0;
   179→    const completed = projectCargo.filter((c) => c.status === 'DISPATCHED' || c.status === 'DELIVERED').length;
   180→    return Math.round((completed / projectCargo.length) * 100);
   181→  }, [projectCargo]);
   182→
   183→  // Donut chart data
   184→  const donutData = useMemo(() => {
   185→    const entries = Object.entries(cargoStatusCounts).sort((a, b) => b[1] - a[1]);
   186→    const total = entries.reduce((s, [, c]) => s + c, 0);
   187→    let cumulative = 0;
   188→    return entries.map(([status, count]) => {
   189→      const start = (cumulative / total) * 100;
   190→      cumulative += count;
   191→      const end = (cumulative / total) * 100;
   192→      const color = CARGO_STATUS_COLORS[status]?.donut || '#64748b';
   193→      return { status, count, start, end, color };
   194→    });
   195→  }, [cargoStatusCounts]);
   196→
   197→  const donutGradient = useMemo(() => {
   198→    if (!Array.isArray(donutData) || donutData.length === 0) return '';
   199→    return donutData.map((d) => `${d.color} ${d.start}% ${d.end}%`).join(', ');
   200→  }, [donutData]);
   201→
   202→  // ETD/ETA timeline calculations
   203→  const etdEtaInfo = useMemo(() => {
   204→    if (!selectedProject?.etd || !selectedProject?.eta) return null;
   205→    const etd = new Date(selectedProject.etd).getTime();

   208→    const totalDays = Math.max(1, Math.ceil((eta - etd) / (1000 * 60 * 60 * 24)));
   209→    const elapsed = Math.ceil((now - etd) / (1000 * 60 * 60 * 24));
   210→    const pct = Math.min(100, Math.max(0, (elapsed / totalDays) * 100));
   211→    const remaining = Math.max(0, Math.ceil((eta - now) / (1000 * 60 * 60 * 24)));
   212→    return { totalDays, elapsed, pct, remaining, isOverdue: now > eta };
   213→  }, [selectedProject]);
   214→
   215→  return (
   216→    <div className="space-y-6">
   217→      {/* Page Header */}
   218→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   219→        <div>
   220→          <h1 className="text-2xl font-bold dark:text-slate-100 text-slate-900">{t('projects.title')}</h1>
   221→          <p className="text-sm dark:text-slate-500 text-slate-400 mt-1">{t('projects.subtitle')}</p>
   222→        </div>
   223→        <Button onClick={() => { setForm(emptyForm); setShowAdd(true); }} className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium shadow-lg shadow-amber-500/20 transition-all duration-200 hover:shadow-xl hover:shadow-amber-500/30">
   224→          <Plus className="h-4 w-4 ml-2" /> {t('projects.addProject')}
   225→        </Button>
   226→      </div>
   227→
   228→      {/* Status Filter Tabs */}
   229→      <Tabs value={statusFilter} onValueChange={setStatusFilter}>
   230→        <TabsList className="dark:bg-slate-800/50 bg-slate-100 dark:border-slate-700/50 border border-slate-200 h-auto p-1 flex-wrap gap-1">
   231→          {statusTabsKeys.map((tab) => (
   232→            <TabsTrigger key={tab.value} value={tab.value} className="data-[state=active]:bg-amber-500/15 data-[state=active]:text-amber-500 dark:text-slate-400 text-slate-500 text-xs px-3 py-1.5">{t(tab.key)}</TabsTrigger>
   233→          ))}
   234→        </TabsList>
   235→      </Tabs>
   236→
   237→      {/* ========== PROJECT DETAIL VIEW ========== */}
   238→      {selectedProject ? (
   239→        <div className="space-y-6">
   240→          {/* Back Button */}
   241→          <Button variant="ghost" onClick={() => setSelectedProject(null)} className="dark:text-slate-400 text-slate-500 hover:dark:text-slate-200 hover:text-slate-900 -mr-2 transition-colors duration-200">
   242→            <ArrowLeft className="h-4 w-4 ml-2" /> {t('projects.backToProjects')}
   243→          </Button>
   244→
   245→          {/* ===== Hero Header ===== */}
   246→          <Card className="dark:border-slate-800 border-slate-200 dark:bg-slate-900/50 bg-white overflow-hidden">
   247→            {/* Gradient top accent bar */}
   248→            <div className={`h-1.5 w-full ${STATUS_COLORS[selectedProject.status].accent}`} />
   249→            <CardContent className="p-6 pt-5">
   250→              <div className="flex flex-col gap-5">
   251→                {/* Top row: avatar, info, actions */}
   252→                <div className="flex flex-col sm:flex-row sm:items-start sm:justify-between gap-4">
   253→                  <div className="flex items-start gap-4">
   254→                    {/* Client avatar */}
   255→                    <div className={`h-14 w-14 rounded-2xl flex items-center justify-center text-xl font-bold shrink-0 ${STATUS_COLORS[selectedProject.status].bg} ${STATUS_COLORS[selectedProject.status].text}`}>
   256→                      {selectedProject.clientName?.charAt(0)?.toUpperCase() || 'P'}
   257→                    </div>
   258→                    <div className="min-w-0">
   259→                      <div className="flex items-center gap-3 flex-wrap">
   260→                        <h2 className="text-xl font-bold dark:text-slate-100 text-slate-900">{selectedProject.name}</h2>
   261→                        <Badge variant="outline" className={`shrink-0 text-xs ${STATUS_COLORS[selectedProject.status].bg} ${STATUS_COLORS[selectedProject.status].text} ${STATUS_COLORS[selectedProject.status].border}`}>
   262→                          {translateStatus(selectedProject.status)}
   263→                        </Badge>
   264→                      </div>
   265→                      <p className="text-sm font-mono dark:text-slate-500 text-slate-400 mt-0.5">{selectedProject.projectCode}</p>
   266→                      {selectedProject.description && <p className="text-sm dark:text-slate-400 text-slate-500 mt-2 max-w-xl leading-relaxed">{selectedProject.description}</p>}

   268→                  </div>

   272→                      <Pencil className="h-3.5 w-3.5 ml-1.5" /> {t('detail.project.editProject')}

   274→                    <Button variant="outline" size="sm" className="dark:border-slate-700 border-slate-300 dark:text-slate-300 text-slate-600 dark:hover:bg-slate-800 hover:bg-slate-100 transition-all duration-200">

   277→                  </div>
   278→                </div>

   280→                {/* Shipping Info Row */}

   282→                  <div className="flex items-center gap-2.5">

   285→                    </div>
   286→                    <div>
   287→                      <p className="text-[10px] uppercase tracking-wider dark:text-slate-600 text-slate-400">{t('detail.project.overview.client')}</p>
   288→                      <p className="dark:text-slate-200 text-slate-800 font-medium">{selectedProject.clientName}</p>
   289→                    </div>
   290→                  </div>
   291→                  <div className="flex items-center gap-2.5">
   292→                    <div className="h-8 w-8 rounded-lg dark:bg-amber-500/10 bg-amber-500/10 flex items-center justify-center">
   293→                      <Ship className="h-4 w-4 dark:text-amber-400 text-amber-600" />
