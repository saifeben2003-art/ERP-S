     1→'use client';
     2→
     3→import { useEffect, useState, useCallback } from 'react';
     4→import { Plus, Search, Pencil, Eye, Trash2, ArrowRight, MapPin, Package, Weight, Move, CheckCircle2, Clock, Truck, Warehouse, CircleDot, ChevronLeft, ChevronRight, Loader2, X } from 'lucide-react';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Label } from '@/components/ui/label';
     9→import { Textarea } from '@/components/ui/textarea';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Skeleton } from '@/components/ui/skeleton';
    12→import { Separator } from '@/components/ui/separator';
    13→import { ScrollArea } from '@/components/ui/scroll-area';
    14→import {
    15→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    16→} from '@/components/ui/table';
    17→import {
    18→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter,
    19→} from '@/components/ui/dialog';
    20→import {
    21→  Sheet, SheetContent, SheetHeader, SheetTitle,
    22→} from '@/components/ui/sheet';
    23→import {
    24→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    25→} from '@/components/ui/select';
    26→import { toast } from 'sonner';
    27→import { useTranslation, translateStatus, translateCategory, translateCommodity, translateMovementType } from '@/lib/translations';
    28→import type { CargoItem, LiftCategory, CommodityType, CargoStatus, Location, Project, Movement } from '@/types/wms';
    29→
    30→const statusStyles: Record<CargoStatus, string> = {
    31→  IN_YARD: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20',
    32→  IN_TRANSIT: 'bg-amber-500/10 text-amber-400 border-amber-500/20',
    33→  DISPATCHED: 'bg-slate-500/10 text-slate-400 border-slate-500/20',
    34→  RECEIVED: 'bg-cyan-500/10 text-cyan-400 border-cyan-500/20',
    35→  IN_WAREHOUSE: 'bg-teal-500/10 text-teal-400 border-teal-500/20',
    36→  DELIVERED: 'bg-slate-400/10 text-slate-500 border-slate-400/20',
    37→};
    38→
    39→const statusIcons: Record<CargoStatus, React.ElementType> = {
    40→  IN_TRANSIT: Truck,
    41→  RECEIVED: CheckCircle2,
    42→  IN_YARD: MapPin,
    43→  IN_WAREHOUSE: Warehouse,
    44→  DISPATCHED: CircleDot,
    45→  DELIVERED: CheckCircle2,
    46→};
    47→
    48→const STATUS_WORKFLOW: Record<CargoStatus, CargoStatus[]> = {
    49→  IN_TRANSIT: ['RECEIVED'],
    50→  RECEIVED: ['IN_YARD', 'IN_WAREHOUSE'],
    51→  IN_YARD: ['IN_WAREHOUSE', 'DISPATCHED'],
    52→  IN_WAREHOUSE: ['IN_YARD', 'DISPATCHED'],
    53→  DISPATCHED: ['DELIVERED'],
    54→  DELIVERED: [],
    55→};
    56→
    57→const ALL_STATUSES: CargoStatus[] = ['IN_TRANSIT', 'RECEIVED', 'IN_YARD', 'IN_WAREHOUSE', 'DISPATCHED', 'DELIVERED'];
    58→
    59→const categoryStyles: Record<LiftCategory, string> = {
    60→  HEAVY_LIFT: 'bg-red-500/10 text-red-400 border-red-500/20',
    61→  OVERSIZE: 'bg-orange-500/10 text-orange-400 border-orange-500/20',
    62→  STANDARD: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20',
    63→  PROJECT_CARGO: 'bg-amber-500/10 text-amber-400 border-amber-500/20',
    64→};
    65→
    66→const movementTypeStyles: Record<string, string> = {
    67→  RECEIVE: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20',
    68→  MOVE: 'bg-cyan-500/10 text-cyan-400 border-cyan-500/20',
    69→  DISPATCH: 'bg-orange-500/10 text-orange-400 border-orange-500/20',
    70→  INSPECT: 'bg-purple-500/10 text-purple-400 border-purple-500/20',
    71→};
    72→
    73→interface CargoListResponse {
    74→  items: CargoItem[];
    75→  total: number;
    76→  page: number;
    77→  totalPages: number;
    78→}
    79→
    80→interface CargoDetail extends CargoItem {
    81→  movements?: Movement[];
    82→}
    83→
    84→const emptyForm = {
    85→  description: '', weight: '', length: '', width: '', height: '',
    86→  liftCategory: '' as string, commodityType: '' as string,
    87→  specialHandling: '', clientName: '', poReference: '', blReference: '',
    88→  centerOfGravity: '', liftingPoints: '', projectId: '',
    89→};
    90→
    91→export function CargoPage() {
    92→  const [cargo, setCargo] = useState<CargoItem[]>([]);
    93→  const [totalPages, setTotalPages] = useState(1);
    94→  const [page, setPage] = useState(1);
    95→  const [loading, setLoading] = useState(true);
    96→  const [search, setSearch] = useState('');
    97→  const [statusFilter, setStatusFilter] = useState('');
    98→  const [categoryFilter, setCategoryFilter] = useState('');
    99→  const [commodityFilter, setCommodityFilter] = useState('');
   100→  const [showAdd, setShowAdd] = useState(false);
   101→  const [editing, setEditing] = useState<CargoItem | null>(null);
   102→  const [deleting, setDeleting] = useState<CargoItem | null>(null);
   103→  const [form, setForm] = useState(emptyForm);
   104→  const [submitting, setSubmitting] = useState(false);
   105→  const [locations, setLocations] = useState<Location[]>([]);
   106→  const [projects, setProjects] = useState<Project[]>([]);
   107→
   108→  // Detail sheet state
   109→  const [detailOpen, setDetailOpen] = useState(false);
   110→  const [detailCargo, setDetailCargo] = useState<CargoDetail | null>(null);
   111→  const [detailLoading, setDetailLoading] = useState(false);
   112→  const [statusChanging, setStatusChanging] = useState(false);
   113→  const [assigningLocation, setAssigningLocation] = useState(false);
   114→  const [selectedLocationId, setSelectedLocationId] = useState('');
   115→
   116→  const { t } = useTranslation();
   117→
   118→  const fetchCargo = useCallback(async () => {
   119→    setLoading(true);
   120→    const params = new URLSearchParams({
   121→      page: String(page), limit: '20',
   122→      ...(search && { search }),
   123→      ...(statusFilter && { status: statusFilter }),
   124→      ...(categoryFilter && { liftCategory: categoryFilter }),
   125→      ...(commodityFilter && { commodityType: commodityFilter }),
   126→    });
   127→    try {
   128→      const res = await fetch(`/api/cargo?${params}`);
   129→      const data: CargoListResponse = await res.json();
   130→      if (!res.ok) { setCargo([]); return; }
   131→      setCargo(data.items || []);
   132→      setTotalPages(data.totalPages);
   133→    } catch { toast.error(t('cargo.toast.fetchFailed')); } finally { setLoading(false); }
   134→  }, [page, search, statusFilter, categoryFilter, commodityFilter, t]);
   135→
   136→  const fetchLookups = useCallback(async () => {
   137→    try {
   138→      const [locRes, projRes] = await Promise.all([fetch('/api/locations?limit=100'), fetch('/api/projects?limit=100')]);
   139→      const locData = await locRes.json();
   140→      const projData = await projRes.json();
   141→      setLocations(locData.items || []);
   142→      setProjects(projData.items || []);
   143→    } catch { /* silent */ }
   144→  }, []);
   145→
   146→  useEffect(() => { fetchCargo(); }, [fetchCargo]);
   147→  useEffect(() => { fetchLookups(); }, [fetchLookups]);
   148→  useEffect(() => { setPage(1); }, [search, statusFilter, categoryFilter, commodityFilter]);
   149→
   150→  const handleSubmit = async () => {
   151→    setSubmitting(true);
   152→    const payload: Record<string, unknown> = {
   153→      description: form.description, weight: parseFloat(form.weight) || 0,
   154→      length: parseFloat(form.length) || 0, width: parseFloat(form.width) || 0, height: parseFloat(form.height) || 0,
   155→      liftCategory: form.liftCategory, commodityType: form.commodityType,
   156→      specialHandling: form.specialHandling || null, clientName: form.clientName || null,
   157→      poReference: form.poReference || null, blReference: form.blReference || null,
   158→      centerOfGravity: form.centerOfGravity || null, liftingPoints: form.liftingPoints ? parseInt(form.liftingPoints) : null,
   159→      projectId: form.projectId && form.projectId !== '_none' ? form.projectId : null,
   160→    };
   161→    try {
   162→      const url = editing ? `/api/cargo/${editing.id}` : '/api/cargo';
   163→      const res = await fetch(url, { method: editing ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   164→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   165→      toast.success(editing ? t('cargo.toast.updated') : t('cargo.toast.created'));
   166→      setShowAdd(false); setEditing(null); setForm(emptyForm); fetchCargo();
   167→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : t('cargo.toast.saveFailed')); } finally { setSubmitting(false); }
   168→  };
   169→
   170→  const handleDelete = async () => {
   171→    if (!deleting) return;
   172→    try {
   173→      const res = await fetch(`/api/cargo/${deleting.id}`, { method: 'DELETE' });
   174→      if (!res.ok) throw new Error('Failed');
   175→      toast.success(t('cargo.toast.deleted')); setDeleting(null); fetchCargo();
   176→    } catch { toast.error(t('cargo.toast.deleteFailed')); }
   177→  };
   178→
   179→  const openEdit = (item: CargoItem) => {
   180→    setEditing(item);
   181→    setForm({
   182→      description: item.description, weight: String(item.weight), length: String(item.length),
   183→      width: String(item.width), height: String(item.height), liftCategory: item.liftCategory,
   184→      commodityType: item.commodityType, specialHandling: item.specialHandling || '',
   185→      clientName: item.clientName || '', poReference: item.poReference || '', blReference: item.blReference || '',
   186→      centerOfGravity: item.centerOfGravity || '', liftingPoints: item.liftingPoints ? String(item.liftingPoints) : '',
   187→      projectId: item.projectId || '',
   188→    });
   189→  };
   190→
   191→  const openDetail = async (item: CargoItem) => {
   192→    setDetailOpen(true);
   193→    setDetailLoading(true);
   194→    setDetailCargo(null);
   195→    setSelectedLocationId(item.locationId || '');
   196→    try {
   197→      const res = await fetch(`/api/cargo/${item.id}`);
   198→      const data = await res.json();
   199→      if (data.data) {
   200→        setDetailCargo(data.data);
   201→        setSelectedLocationId(data.data.locationId || '');
   202→      } else {
   203→        setDetailCargo(item);
   204→      }
   205→    } catch { setDetailCargo(item); } finally { setDetailLoading(false); }
   206→  };
   207→
   208→  const handleStatusChange = async (newStatus: CargoStatus) => {
   209→    if (!detailCargo) return;
   210→    setStatusChanging(true);
   211→    try {
   212→      const res = await fetch(`/api/cargo/${detailCargo.id}/status`, {
   213→        method: 'PATCH', headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({ status: newStatus }),
   215→      });
   216→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   217→      toast.success(t('detail.statusChanged'));
   218→      openDetail(detailCargo); // refresh
   219→      fetchCargo();
   220→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : t('detail.statusChangeFailed')); } finally { setStatusChanging(false); }
   221→  };
   222→
   223→  const handleAssignLocation = async () => {
   224→    if (!detailCargo || !selectedLocationId) return;
   225→    setAssigningLocation(true);
   226→    try {
   227→      const res = await fetch(`/api/cargo/${detailCargo.id}`, {
   228→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   229→        body: JSON.stringify({ locationId: selectedLocationId }),
   230→      });
   231→      if (!res.ok) throw new Error('Failed');
   232→      toast.success(t('detail.locationAssigned'));
   233→      openDetail(detailCargo);
   234→      fetchCargo();
   235→    } catch { toast.error(t('detail.statusChangeFailed')); } finally { setAssigningLocation(false); }
   236→  };
   237→
   238→  const currentStep = detailCargo ? ALL_STATUSES.indexOf(detailCargo.status) : -1;
   239→
   240→  return (
   241→    <div className="space-y-6">
   242→      {/* Header */}
   243→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   244→        <div>
   245→          <h1 className="text-2xl font-bold text-slate-100">{t('cargo.title')}</h1>
   246→          <p className="text-sm text-slate-500 mt-1">{t('cargo.subtitle')}</p>
   247→        </div>
   248→        <Button onClick={() => { setForm(emptyForm); setShowAdd(true); }} className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium">
   249→          <Plus className="h-4 w-4 ml-2" /> {t('cargo.addCargo')}
   250→        </Button>
   251→      </div>
   252→
   253→      {/* Filters */}
   254→      <Card className="border-slate-800 bg-slate-900/50">
   255→        <CardContent className="p-4">
   256→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   257→            <div className="relative">
   258→              <Search className="absolute right-2.5 top-2.5 h-4 w-4 text-slate-500" />
   259→              <Input placeholder={t('cargo.searchPlaceholder')} value={search} onChange={(e) => setSearch(e.target.value)}
   260→                className="border-slate-700 bg-slate-800 pr-9 text-slate-200 placeholder:text-slate-600" />
   261→            </div>
   262→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   263→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300"><SelectValue placeholder={t('common.allStatuses')} /></SelectTrigger>
   264→              <SelectContent className="border-slate-700 bg-slate-800">
   265→                {ALL_STATUSES.map((s) => (<SelectItem key={s} value={s} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">{translateStatus(s)}</SelectItem>))}
   266→              </SelectContent>
   267→            </Select>
   268→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   269→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300"><SelectValue placeholder={t('common.allCategories')} /></SelectTrigger>
   270→              <SelectContent className="border-slate-700 bg-slate-800">
   271→                {['HEAVY_LIFT', 'OVERSIZE', 'STANDARD', 'PROJECT_CARGO'].map((c) => (<SelectItem key={c} value={c} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">{translateCategory(c)}</SelectItem>))}
   272→              </SelectContent>
   273→            </Select>
   274→            <Select value={commodityFilter} onValueChange={setCommodityFilter}>
   275→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300"><SelectValue placeholder={t('common.allCommodities')} /></SelectTrigger>
   276→              <SelectContent className="border-slate-700 bg-slate-800">
   277→                {['GENERAL', 'MACHINERY', 'STEEL', 'EQUIPMENT', 'MODULE'].map((c) => (<SelectItem key={c} value={c} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">{translateCommodity(c)}</SelectItem>))}
   278→              </SelectContent>
   279→            </Select>
   280→          </div>
   281→        </CardContent>
   282→      </Card>
   283→
   284→      {/* Table */}
   285→      <Card className="border-slate-800 bg-slate-900/50">
   286→        <CardContent className="p-0">
   287→          <div className="overflow-x-auto">
   288→            <Table>
   289→              <TableHeader>
   290→                <TableRow className="border-slate-800 hover:bg-transparent">
   291→                  <TableHead className="text-xs text-slate-500">{t('cargo.table.code')}</TableHead>
   292→                  <TableHead className="text-xs text-slate-500">{t('cargo.table.description')}</TableHead>
   293→                  <TableHead className="text-xs text-slate-500 hidden md:table-cell">{t('cargo.table.weight')}</TableHead>
   294→                  <TableHead className="text-xs text-slate-500 hidden lg:table-cell">{t('cargo.table.category')}</TableHead>
   295→                  <TableHead className="text-xs text-slate-500">{t('cargo.table.status')}</TableHead>
   296→                  <TableHead className="text-xs text-slate-500 hidden lg:table-cell">{t('cargo.table.location')}</TableHead>
   297→                  <TableHead className="text-xs text-slate-500 text-left">{t('common.actions')}</TableHead>
   298→                </TableRow>
   299→              </TableHeader>
   300→              <TableBody>
   301→                {loading ? Array.from({ length: 5 }).map((_, i) => (
   302→                  <TableRow key={i} className="border-slate-800 hover:bg-transparent">
   303→                    {Array.from({ length: 7 }).map((_, j) => (<TableCell key={j} className="py-3"><Skeleton className="h-4 w-16 bg-slate-800" /></TableCell>))}
   304→                  </TableRow>
   305→                )) : cargo.length === 0 ? (
   306→                  <TableRow className="border-slate-800 hover:bg-transparent"><TableCell colSpan={7} className="text-center py-8 text-slate-500">{t('cargo.noCargoFound')}</TableCell></TableRow>
   307→                ) : cargo.map((item) => (
   308→                  <TableRow key={item.id} className="border-slate-800 hover:bg-slate-800/50 cursor-pointer" onClick={() => openDetail(item)}>
   309→                    <TableCell className="py-3 text-xs font-mono font-medium text-amber-400/80">{item.cargoCode}</TableCell>
   310→                    <TableCell className="py-3 text-xs text-slate-300 max-w-[200px] truncate">{item.description}</TableCell>
   311→                    <TableCell className="py-3 text-xs text-slate-400 hidden md:table-cell">{item.weight.toLocaleString()}</TableCell>
   312→                    <TableCell className="py-3 hidden lg:table-cell">
   313→                      <Badge variant="outline" className={`text-[10px] ${categoryStyles[item.liftCategory]}`}>{translateCategory(item.liftCategory)}</Badge>
   314→                    </TableCell>
   315→                    <TableCell className="py-3">
   316→                      <Badge variant="outline" className={`text-[10px] ${statusStyles[item.status]}`}>{translateStatus(item.status)}</Badge>
   317→                    </TableCell>
   318→                    <TableCell className="py-3 text-xs text-slate-400 hidden lg:table-cell">{item.location?.code || '—'}</TableCell>
   319→                    <TableCell className="py-3 text-left" onClick={(e) => e.stopPropagation()}>
   320→                      <div className="flex items-center justify-start gap-1">
   321→                        <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-cyan-400" onClick={() => openDetail(item)}><Eye className="h-3.5 w-3.5" /></Button>
   322→                        <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-amber-400" onClick={() => openEdit(item)}><Pencil className="h-3.5 w-3.5" /></Button>
   323→                        <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-red-400" onClick={() => setDeleting(item)}><Trash2 className="h-3.5 w-3.5" /></Button>
   324→                      </div>
   325→                    </TableCell>
   326→                  </TableRow>
   327→                ))}
   328→              </TableBody>
   329→            </Table>
   330→          </div>
   331→        </CardContent>
   332→      </Card>
   333→
   334→      {/* Pagination */}
   335→      {totalPages > 1 && (
   336→        <div className="flex items-center justify-center gap-2">
   337→          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(page - 1)} className="border-slate-700 bg-slate-800 text-slate-300 hover:bg-slate-700">{t('common.previous')}</Button>
   338→          <span className="text-xs text-slate-500">{t('common.page')} {page} {t('common.of')} {totalPages}</span>
   339→          <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage(page + 1)} className="border-slate-700 bg-slate-800 text-slate-300 hover:bg-slate-700">{t('common.next')}</Button>
   340→        </div>
   341→      )}
   342→
   343→      {/* ========== DETAIL SHEET ========== */}
   344→      <Sheet open={detailOpen} onOpenChange={setDetailOpen}>
   345→        <SheetContent side="left" className="w-full sm:max-w-xl border-slate-700 bg-slate-900 p-0 overflow-hidden flex flex-col">
   346→          {detailLoading ? (
   347→            <div className="flex-1 flex items-center justify-center"><Loader2 className="h-8 w-8 animate-spin text-amber-400" /></div>
   348→          ) : detailCargo ? (
   349→            <>
   350→              <SheetHeader className="p-4 border-b border-slate-800 shrink-0">
   351→                <div className="flex items-center justify-between">
   352→                  <SheetTitle className="text-sm font-mono text-amber-400">{detailCargo.cargoCode}</SheetTitle>
   353→                  <Badge variant="outline" className={`text-[10px] ${statusStyles[detailCargo.status]}`}>{translateStatus(detailCargo.status)}</Badge>
   354→                </div>
   355→                <p className="text-sm text-slate-200 mt-1 font-medium">{detailCargo.description}</p>
   356→              </SheetHeader>
   357→
   358→              <ScrollArea className="flex-1">
   359→                <div className="p-4 space-y-5">
   360→                  {/* Status Workflow Stepper */}
   361→                  <div className="rounded-xl border border-slate-800 bg-slate-900/80 p-4">
   362→                    <h3 className="text-xs font-semibold text-slate-400 mb-3">{t('detail.workflow.title')}</h3>
   363→                    <div className="flex items-center gap-1 overflow-x-auto pb-1">
   364→                      {ALL_STATUSES.map((s, idx) => {
   365→                        const Icon = statusIcons[s];
   366→                        const isCompleted = idx <= currentStep;
   367→                        const isCurrent = idx === currentStep;
   368→                        const isNext = STATUS_WORKFLOW[detailCargo.status]?.includes(s);
   369→                        return (
   370→                          <div key={s} className="flex items-center shrink-0">
   371→                            <button
   372→                              disabled={!isNext || statusChanging}
   373→                              onClick={() => isNext && handleStatusChange(s)}
   374→                              className={`flex flex-col items-center gap-1.5 px-2 py-1.5 rounded-lg transition-all min-w-[56px] ${
   375→                                isCurrent ? 'bg-amber-500/15 ring-1 ring-amber-500/40' :
   376→                                isCompleted ? 'opacity-60' :
   377→                                isNext ? 'hover:bg-slate-800 cursor-pointer' : 'opacity-30'
   378→                              }`}
   379→                            >
   380→                              <Icon className={`h-4 w-4 ${isCurrent ? 'text-amber-400' : isCompleted ? 'text-emerald-400' : isNext ? 'text-slate-400' : 'text-slate-600'}`} />
   381→                              <span className={`text-[9px] leading-tight text-center ${isCurrent ? 'text-amber-400 font-semibold' : 'text-slate-500'}`}>{translateStatus(s)}</span>
   382→                            </button>
   383→                            {idx < ALL_STATUSES.length - 1 && (
   384→                              <ChevronLeft className={`h-3 w-3 mx-0.5 shrink-0 ${idx < currentStep ? 'text-emerald-400' : 'text-slate-700'}`} />
   385→                            )}
   386→                          </div>
   387→                        );
   388→                      })}
   389→                    </div>
   390→                  </div>
   391→
   392→                  {/* Specifications */}
   393→                  <div className="rounded-xl border border-slate-800 bg-slate-900/80 p-4">
