  1040→export function useTranslation() {
  1041→  const locale = useAppStore((s) => s.locale);
  1042→  const tr = useCallback((key: string): string => {
  1043→    return localeMap[locale]?.[key] || key;
  1044→  }, [locale]);
  1045→
  1046→  return { t: tr, locale };
