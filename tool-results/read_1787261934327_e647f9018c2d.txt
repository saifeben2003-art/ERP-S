    68→function loadPct(loc: Location) {
    69→  if (!loc.maxWeight || loc.maxWeight === 0) return 0;
    70→  return Math.min(100, Math.round((loc.currentLoad / loc.maxWeight) * 100));
    71→}
    72→
    73→function gaugeColor(pct: number) {
    74→  if (pct >= 80) return { bar: 'bg-red-500', text: 'text-red-600 dark:text-red-400', ring: 'stroke-red-500' };
    75→  if (pct >= 50) return { bar: 'bg-amber-500', text: 'text-amber-600 dark:text-amber-400', ring: 'stroke-amber-500' };
    76→  return { bar: 'bg-emerald-500', text: 'text-emerald-600 dark:text-emerald-400', ring: 'stroke-emerald-500' };
    77→}
    78→
    79→function capacityBarColor(pct: number) {
    80→  if (pct >= 80) return 'bg-red-500';
    81→  if (pct >= 50) return 'bg-amber-500';
    82→  return 'bg-emerald-500';
    83→}

   101→function OccupancyGauge({ percentage }: { percentage: number }) {
   102→  const colors = gaugeColor(percentage);
   103→  const circumference = 2 * Math.PI * 40;
   104→  const strokeDashoffset = circumference - (percentage / 100) * circumference;
   105→  return (
   106→    <div className="relative flex items-center justify-center">
   107→      <svg width="100" height="100" viewBox="0 0 100 100" className="-rotate-90">
   108→        <circle cx="50" cy="50" r="40" fill="none" strokeWidth="8" className="stroke-slate-200 dark:stroke-slate-700" />
   109→        <circle cx="50" cy="50" r="40" fill="none" strokeWidth="8" strokeLinecap="round"
   110→          className={cn('transition-all duration-700', colors.ring)}
   111→          strokeDasharray={circumference} strokeDashoffset={strokeDashoffset} />
   112→      </svg>
   113→      <div className="absolute flex flex-col items-center">
   114→        <span className={cn('text-2xl font-bold', colors.text)}>{percentage}%</span>
   115→        <span className="text-[10px] text-slate-500 dark:text-slate-400">{percentage >= 80 ? 'Critical' : percentage >= 50 ? 'Warning' : 'Normal'}</span>
   116→      </div>
   117→    </div>
   118→  );
   119→}

   166→export function LocationsPage() {
   167→  const [locations, setLocations] = useState<Location[]>([]);
   168→  const [loading, setLoading] = useState(true);
   169→  const [typeFilter, setTypeFilter] = useState('');
   170→  const [searchQuery, setSearchQuery] = useState('');
   171→  const [viewMode, setViewMode] = useState<ViewMode>('table');
   172→  const [showAdd, setShowAdd] = useState(false);
   173→  const [editing, setEditing] = useState<Location | null>(null);
   174→  const [deleting, setDeleting] = useState<Location | null>(null);
   175→  const [form, setForm] = useState(emptyForm);
   176→  const [submitting, setSubmitting] = useState(false);
   177→  const [detailOpen, setDetailOpen] = useState(false);
   178→  const [detailLocation, setDetailLocation] = useState<Location | null>(null);
   179→  const [detailCargo, setDetailCargo] = useState<CargoItem[]>([]);
   180→  const [detailLoading, setDetailLoading] = useState(false);
   181→  const [selectedRows, setSelectedRows] = useState<Set<string>>(new Set());
   182→  const { t } = useTranslation();
   183→
   184→  const fetchLocations = useCallback(async () => {
   185→    setLoading(true);
   186→    const params = new URLSearchParams({ limit: '100', ...(typeFilter && { type: typeFilter }) });
   187→    try {
   188→      const res = await fetch('/api/locations?' + params);
   189→      const data: LocationListResponse = await res.json();
   190→      if (!res.ok) { setLocations([]); return; }
   191→      setLocations(data.items || []);
   192→    } catch { toast.error(t('locations.toast.fetchFailed')); } finally { setLoading(false); }
   193→  }, [typeFilter, t]);
   194→
   195→  useEffect(() => { fetchLocations(); }, [fetchLocations]);
   196→
   197→  const filteredLocations = locations.filter((loc) => {
   198→    if (!searchQuery) return true;
   199→    const q = searchQuery.toLowerCase();
   200→    return loc.code.toLowerCase().includes(q) || loc.name.toLowerCase().includes(q) || (loc.zone && loc.zone.toLowerCase().includes(q));
   201→  });
   202→
   203→  const handleSubmit = async () => {
   204→    setSubmitting(true);
   205→    const payload: Record<string, unknown> = {
   206→      code: form.code, name: form.name, type: form.type, zone: form.zone || null,
   207→      maxWeight: form.maxWeight ? parseFloat(form.maxWeight) : null,
   208→      maxDimension: form.maxDimension || null, area: form.area ? parseFloat(form.area) : null, isActive: form.isActive,
   209→    };
   210→    try {
   211→      const url = editing ? '/api/locations/' + editing.id : '/api/locations';
   212→      const res = await fetch(url, { method: editing ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   213→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   214→      toast.success(editing ? t('locations.toast.updated') : t('locations.toast.created'));
   215→      setShowAdd(false); setEditing(null); setForm(emptyForm); fetchLocations();
   216→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : 'Failed'); } finally { setSubmitting(false); }
   217→  };
   218→
   219→  const handleDelete = async () => {
   220→    if (!deleting) return;
   221→    try {
   222→      const res = await fetch('/api/locations/' + deleting.id, { method: 'DELETE' });
   223→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   224→      toast.success(t('locations.toast.deleted')); setDeleting(null); fetchLocations();
   225→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : t('locations.toast.deleteFailed')); }
   226→  };
   227→
   228→  const handleToggleActive = async (loc: Location) => {
   229→    try {
   230→      const res = await fetch('/api/locations/' + loc.id, {
   231→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({ isActive: !loc.isActive }),
   233→      });
   234→      if (!res.ok) throw new Error('Failed');
   235→      toast.success(loc.isActive ? t('common.disabled') : t('common.enabled'));
   236→      fetchLocations();
   237→      if (detailLocation?.id === loc.id) {
   238→        setDetailLocation({ ...loc, isActive: !loc.isActive });
   239→      }
   240→    } catch { toast.error('Failed to update status'); }
   241→  };
   242→
   243→  const openEdit = (loc: Location) => {
   244→    setEditing(loc);
   245→    setForm({ code: loc.code, name: loc.name, type: loc.type, zone: loc.zone || '', maxWeight: loc.maxWeight ? String(loc.maxWeight) : '', maxDimension: loc.maxDimension || '', area: loc.area ? String(loc.area) : '', isActive: loc.isActive });
   246→  };
   247→
   248→  const openDetail = async (loc: Location) => {
   249→    setDetailLocation(loc); setDetailOpen(true); setDetailLoading(true);
   250→    try {
   251→      const res = await fetch('/api/cargo?locationId=' + loc.id + '&limit=200');
   252→      const data = await res.json();
   253→      setDetailCargo(data.items || []);
   254→    } catch { setDetailCargo([]); } finally { setDetailLoading(false); }
   255→  };
   256→
   257→  const toggleSelectRow = (id: string) => {
   258→    setSelectedRows((prev) => {
   259→      const next = new Set(prev);
   260→      if (next.has(id)) next.delete(id); else next.add(id);
   261→      return next;
   262→    });
   263→  };
   264→
   265→  const toggleSelectAll = () => {
   266→    const fLocs = Array.isArray(filteredLocations) ? filteredLocations : [];
   267→    if (selectedRows.size === fLocs.length) {
   268→      setSelectedRows(new Set());
   269→    } else {
   270→      setSelectedRows(new Set(fLocs.map((l) => l.id)));
   271→    }
   272→  };
   273→
   274→  const handleBulkDelete = async () => {
   275→    try {
   276→      await Promise.all(Array.from(selectedRows).map((id) => fetch('/api/locations/' + id, { method: 'DELETE' })));
   277→      toast.success(t('locations.toast.deleted'));
   278→      setSelectedRows(new Set());
   279→      fetchLocations();
   280→    } catch { toast.error(t('locations.toast.deleteFailed')); }
   281→  };
   282→
   283→  // ==================== RENDER ====================
   284→
   285→  return (
   286→    <div className="space-y-6">
   287→      {/* ===== Header ===== */}
   288→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   289→        <div>
   290→          <h1 className="text-2xl font-bold text-slate-900 dark:text-slate-100">{t('locations.title')}</h1>
   291→          <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">{t('locations.subtitle')}</p>
   292→        </div>
   293→        <Button onClick={() => { setForm(emptyForm); setShowAdd(true); }} className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium transition-all duration-200 hover:shadow-md">
   294→          <Plus className="h-4 w-4 ml-2" /> {t('locations.addLocation')}
   295→        </Button>
   296→      </div>
   297→
   298→      {/* ===== Filters & Controls ===== */}
   299→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   300→        <div className="flex flex-wrap gap-2">
   301→          {locationTypes.map((lt) => (
   302→            <Button key={lt} variant={typeFilter === lt ? 'default' : 'outline'} size="sm" onClick={() => setTypeFilter(typeFilter === lt ? '' : lt)}
   303→              className={cn(
   304→                'transition-all duration-200',
   305→                typeFilter === lt
   306→                  ? 'bg-amber-500/15 text-amber-600 dark:text-amber-400 border-amber-500/30 hover:bg-amber-500/20'
   307→                  : 'border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900/50 text-slate-500 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800 hover:text-slate-700 dark:hover:text-slate-300'
   308→              )}>
   309→              {translateLocationType(lt)}
   310→            </Button>
   311→          ))}
   312→          <Button variant={typeFilter === '' ? 'default' : 'outline'} size="sm" onClick={() => setTypeFilter('')}
   313→            className={cn(
   314→              'transition-all duration-200',
   315→              typeFilter === ''
   316→                ? 'bg-amber-500/15 text-amber-600 dark:text-amber-400 border-amber-500/30 hover:bg-amber-500/20'
   317→                : 'border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900/50 text-slate-500 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800 hover:text-slate-700 dark:hover:text-slate-300'
   318→            )}>
   319→            {t('locations.allTypes')}
   320→          </Button>
   321→        </div>
   322→        <div className="flex items-center gap-2">
   323→          {/* Search */}
   324→          <div className="relative">
   325→            <Search className="absolute start-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-slate-400 dark:text-slate-500" />
   326→            <Input
   327→              placeholder={t('locations.searchPlaceholder')}
   328→              value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)}
   329→              className="h-9 w-48 bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-700 text-slate-900 dark:text-slate-200 ps-8 text-sm"
   330→            />
   331→          </div>
   332→          {/* View Toggle */}
   333→          <div className="flex rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900/50 p-0.5">
   334→            <button
   335→              onClick={() => setViewMode('table')}
   336→              className={cn(
   337→                'flex items-center gap-1.5 rounded-md px-2.5 py-1.5 text-xs font-medium transition-all duration-200',
   338→                viewMode === 'table'
   339→                  ? 'bg-amber-500/15 text-amber-600 dark:text-amber-400'
   340→                  : 'text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-300'
   341→              )}
   342→            >
   343→              <LayoutList className="h-3.5 w-3.5" /> {t('locations.tableView')}
   344→            </button>
   345→            <button
   346→              onClick={() => setViewMode('grid')}
   347→              className={cn(
   348→                'flex items-center gap-1.5 rounded-md px-2.5 py-1.5 text-xs font-medium transition-all duration-200',
   349→                viewMode === 'grid'
   350→                  ? 'bg-amber-500/15 text-amber-600 dark:text-amber-400'
   351→                  : 'text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-300'
   352→              )}
   353→            >
   354→              <LayoutGrid className="h-3.5 w-3.5" /> {t('locations.gridView')}
   355→            </button>
   356→          </div>
   357→        </div>
   358→      </div>
   359→
   360→      {/* ===== Bulk Actions Bar ===== */}
   361→      {selectedRows.size > 0 && (
   362→        <div className="flex items-center gap-3 rounded-lg border border-amber-500/20 bg-amber-500/5 dark:bg-amber-500/5 px-4 py-2.5">
   363→          <span className="text-sm text-amber-600 dark:text-amber-400 font-medium">
   364→            {selectedRows.size} {t('common.selected')}
   365→          </span>
   366→          <Button variant="outline" size="sm" onClick={handleBulkDelete}
   367→            className="border-red-200 dark:border-red-500/20 text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-500/10 transition-all duration-200">
   368→            <Trash2 className="h-3.5 w-3.5 ml-1.5" /> {t('common.delete')}
   369→          </Button>
   370→          <Button variant="ghost" size="sm" onClick={() => setSelectedRows(new Set())}
   371→            className="text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-200">
   372→            {t('common.clear')}
   373→          </Button>
   374→        </div>
   375→      )}
   376→
   377→      {/* ===== Table View ===== */}
   378→      {viewMode === 'table' && (
   379→        <div className="rounded-xl border border-slate-200 dark:border-slate-800 bg-white dark:bg-slate-900/30 overflow-hidden">
   380→          {loading ? (
   381→            <div className="p-6 space-y-3">
   382→              {Array.from({ length: 6 }).map((_, i) => (
   383→                <Skeleton key={i} className="h-12 w-full bg-slate-100 dark:bg-slate-800 rounded-lg" />
   384→              ))}
   385→            </div>
   386→          ) : !Array.isArray(filteredLocations) || filteredLocations.length === 0 ? (
   387→            <div className="flex flex-col items-center justify-center py-16 text-slate-400 dark:text-slate-500">
   388→              <MapPin className="h-10 w-10 mb-3 text-slate-300 dark:text-slate-600" />
   389→              <p className="text-sm font-medium">{t('detail.location.emptyState')}</p>
   390→            </div>
   391→          ) : (
   392→            <div className="max-h-[520px] overflow-y-auto">
   393→              <Table>
   394→                <TableHeader>
   395→                  <TableRow className="border-slate-200 dark:border-slate-800 hover:bg-transparent">
   396→                    <TableHead className="w-10 ps-4">
   397→                      <Checkbox checked={Array.isArray(filteredLocations) && selectedRows.size === filteredLocations.length && filteredLocations.length > 0} onCheckedChange={toggleSelectAll} />
   398→                    </TableHead>
   399→                    <TableHead className="text-[11px] font-semibold text-slate-500 dark:text-slate-400 uppercase tracking-wider">{t('locations.table.code')}</TableHead>
