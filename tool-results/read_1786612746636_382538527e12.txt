     1→     1→'use client';
     2→     2→
     3→     3→import { useEffect, useState, useCallback } from 'react';
     4→     4→import { Plus, Search, Pencil, Eye, Trash2, X } from 'lucide-react';
     5→     5→import { Card, CardContent } from '@/components/ui/card';
     6→     6→import { Button } from '@/components/ui/button';
     7→     7→import { Input } from '@/components/ui/input';
     8→     8→import { Label } from '@/components/ui/label';
     9→     9→import { Textarea } from '@/components/ui/textarea';
    10→    10→import { Badge } from '@/components/ui/badge';
    11→    11→import { Skeleton } from '@/components/ui/skeleton';
    12→    12→import {
    13→    13→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    14→    14→} from '@/components/ui/table';
    15→    15→import {
    16→    16→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter,
    17→    17→} from '@/components/ui/dialog';
    18→    18→import {

    20→    20→} from '@/components/ui/select';

    22→    22→import type { CargoItem, LiftCategory, CommodityType, CargoStatus, Location, Project } from '@/types/wms';
    23→    23→
    24→    24→const statusStyles: Record<CargoStatus, string> = {
    25→    25→  IN_YARD: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20',
    26→    26→  IN_TRANSIT: 'bg-amber-500/10 text-amber-400 border-amber-500/20',
    27→    27→  DISPATCHED: 'bg-slate-500/10 text-slate-400 border-slate-500/20',
    28→    28→  RECEIVED: 'bg-cyan-500/10 text-cyan-400 border-cyan-500/20',
    29→    29→  IN_WAREHOUSE: 'bg-teal-500/10 text-teal-400 border-teal-500/20',
    30→    30→  DELIVERED: 'bg-slate-400/10 text-slate-500 border-slate-400/20',
    31→    31→};
    32→    32→
    33→    33→const categoryStyles: Record<LiftCategory, string> = {
    34→    34→  HEAVY_LIFT: 'bg-red-500/10 text-red-400 border-red-500/20',
    35→    35→  OVERSIZE: 'bg-orange-500/10 text-orange-400 border-orange-500/20',
    36→    36→  STANDARD: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20',
    37→    37→  PROJECT_CARGO: 'bg-amber-500/10 text-amber-400 border-amber-500/20',
    38→    38→};
    39→    39→
    40→    40→interface CargoListResponse {
    41→    41→  items: CargoItem[];
    42→    42→  total: number;
    43→    43→  page: number;
    44→    44→  totalPages: number;
    45→    45→}
    46→    46→
    47→    47→const emptyForm = {
    48→    48→  description: '', weight: '', length: '', width: '', height: '',
    49→    49→  liftCategory: '' as string, commodityType: '' as string,
    50→    50→  specialHandling: '', clientName: '', poReference: '', blReference: '',
    51→    51→  centerOfGravity: '', liftingPoints: '', projectId: '',
    52→    52→};
    53→    53→
    54→    54→export function CargoPage() {
    55→    55→  const [cargo, setCargo] = useState<CargoItem[]>([]);
    56→    56→  const [totalPages, setTotalPages] = useState(1);
    57→    57→  const [page, setPage] = useState(1);
    58→    58→  const [loading, setLoading] = useState(true);
    59→    59→  const [search, setSearch] = useState('');
    60→    60→  const [statusFilter, setStatusFilter] = useState('');
    61→    61→  const [categoryFilter, setCategoryFilter] = useState('');
    62→    62→  const [commodityFilter, setCommodityFilter] = useState('');
    63→    63→  const [showAdd, setShowAdd] = useState(false);
    64→    64→  const [editing, setEditing] = useState<CargoItem | null>(null);
    65→    65→  const [viewing, setViewing] = useState<CargoItem | null>(null);
    66→    66→  const [deleting, setDeleting] = useState<CargoItem | null>(null);
    67→    67→  const [form, setForm] = useState(emptyForm);
    68→    68→  const [submitting, setSubmitting] = useState(false);
    69→    69→  const [locations, setLocations] = useState<Location[]>([]);
    70→    70→  const [projects, setProjects] = useState<Project[]>([]);
    71→    71→
    72→    72→  const fetchCargo = useCallback(async () => {
    73→    73→    setLoading(true);
    74→    74→    const params = new URLSearchParams({
    75→    75→      page: String(page),
    76→    76→      limit: '10',
    77→    77→      ...(search && { search }),
    78→    78→      ...(statusFilter && { status: statusFilter }),
    79→    79→      ...(categoryFilter && { liftCategory: categoryFilter }),
    80→    80→      ...(commodityFilter && { commodityType: commodityFilter }),
    81→    81→    });
    82→    82→    try {
    83→    83→      const res = await fetch(`/api/cargo?${params}`);
    84→    84→      const data: CargoListResponse = await res.json();
    85→    85→      setCargo(data.items);
    86→    86→      setTotalPages(data.totalPages);
    87→    87→    } catch {
    88→    88→      toast.error('Failed to fetch cargo items');
    89→    89→    } finally {
    90→    90→      setLoading(false);
    91→    91→    }
    92→    92→  }, [page, search, statusFilter, categoryFilter, commodityFilter]);
    93→    93→
    94→    94→  const fetchLookups = useCallback(async () => {
    95→    95→    try {
    96→    96→      const [locRes, projRes] = await Promise.all([
    97→    97→        fetch('/api/locations?limit=100'),
    98→    98→        fetch('/api/projects?limit=100'),
    99→    99→      ]);
   100→   100→      const locData = await locRes.json();
   101→   101→      const projData = await projRes.json();
   102→   102→      setLocations(locData.items || []);
   103→   103→      setProjects(projData.items || []);
   104→   104→    } catch {
   105→   105→      // silent
   106→   106→    }
   107→   107→  }, []);
   108→   108→
   109→   109→  useEffect(() => { fetchCargo(); }, [fetchCargo]);
   110→   110→  useEffect(() => { fetchLookups(); }, [fetchLookups]);
   111→   111→  useEffect(() => { setPage(1); }, [search, statusFilter, categoryFilter, commodityFilter]);
   112→   112→
   113→   113→  const handleSubmit = async () => {
   114→   114→    setSubmitting(true);
   115→   115→    const payload: Record<string, unknown> = {
   116→   116→      description: form.description,
   117→   117→      weight: parseFloat(form.weight) || 0,
   118→   118→      length: parseFloat(form.length) || 0,
   119→   119→      width: parseFloat(form.width) || 0,
   120→   120→      height: parseFloat(form.height) || 0,
   121→   121→      liftCategory: form.liftCategory,
   122→   122→      commodityType: form.commodityType,
   123→   123→      specialHandling: form.specialHandling || null,
   124→   124→      clientName: form.clientName || null,
   125→   125→      poReference: form.poReference || null,
   126→   126→      blReference: form.blReference || null,
   127→   127→      centerOfGravity: form.centerOfGravity || null,
   128→   128→      liftingPoints: form.liftingPoints ? parseInt(form.liftingPoints) : null,
   129→   129→      projectId: form.projectId || null,
   130→   130→    };
   131→   131→
   132→   132→    try {
   133→   133→      const url = editing ? `/api/cargo/${editing.id}` : '/api/cargo';
   134→   134→      const method = editing ? 'PUT' : 'POST';
   135→   135→      const res = await fetch(url, {
   136→   136→        method,
   137→   137→        headers: { 'Content-Type': 'application/json' },
   138→   138→        body: JSON.stringify(payload),
   139→   139→      });
   140→   140→      if (!res.ok) {
   141→   141→        const err = await res.json();
   142→   142→        throw new Error(err.error || 'Failed to save');
   143→   143→      }
   144→   144→      toast.success(editing ? 'Cargo updated' : 'Cargo created');
   145→   145→      setShowAdd(false);
   146→   146→      setEditing(null);
   147→   147→      setForm(emptyForm);
   148→   148→      fetchCargo();
   149→   149→    } catch (e: unknown) {
   150→   150→      toast.error(e instanceof Error ? e.message : 'Failed to save');
   151→   151→    } finally {
   152→   152→      setSubmitting(false);
   153→   153→    }
   154→   154→  };
   155→   155→
   156→   156→  const handleDelete = async () => {
   157→   157→    if (!deleting) return;
   158→   158→    try {
   159→   159→      const res = await fetch(`/api/cargo/${deleting.id}`, { method: 'DELETE' });
   160→   160→      if (!res.ok) throw new Error('Failed to delete');
   161→   161→      toast.success('Cargo deleted');
   162→   162→      setDeleting(null);
   163→   163→      fetchCargo();
   164→   164→    } catch {
   165→   165→      toast.error('Failed to delete cargo');
   166→   166→    }
   167→   167→  };
   168→   168→
   169→   169→  const openEdit = (item: CargoItem) => {
   170→   170→    setEditing(item);
   171→   171→    setForm({
   172→   172→      description: item.description,
   173→   173→      weight: String(item.weight),
   174→   174→      length: String(item.length),
   175→   175→      width: String(item.width),
   176→   176→      height: String(item.height),
   177→   177→      liftCategory: item.liftCategory,
   178→   178→      commodityType: item.commodityType,
   179→   179→      specialHandling: item.specialHandling || '',
   180→   180→      clientName: item.clientName || '',
   181→   181→      poReference: item.poReference || '',
   182→   182→      blReference: item.blReference || '',
   183→   183→      centerOfGravity: item.centerOfGravity || '',
   184→   184→      liftingPoints: item.liftingPoints ? String(item.liftingPoints) : '',
   185→   185→      projectId: item.projectId || '',
   186→   186→    });
   187→   187→  };
   188→   188→
   189→   189→  return (
   190→   190→    <div className="space-y-6">
   191→   191→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   192→   192→        <div>
   193→   193→          <h1 className="text-2xl font-bold text-slate-100">Cargo Items</h1>
   194→   194→          <p className="text-sm text-slate-500 mt-1">Manage all cargo in the warehouse</p>
   195→   195→        </div>
   196→   196→        <Button
   197→   197→          onClick={() => { setForm(emptyForm); setShowAdd(true); }}
   198→   198→          className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium"
   199→   199→        >
   200→   200→          <Plus className="h-4 w-4 mr-2" /> Add Cargo
   201→   201→        </Button>
   202→   202→      </div>
   203→   203→
   204→   204→      {/* Filters */}
   205→   205→      <Card className="border-slate-800 bg-slate-900/50">
   206→   206→        <CardContent className="p-4">
   207→   207→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   208→   208→            <div className="relative">
   209→   209→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-slate-500" />
   210→   210→              <Input
   211→   211→                placeholder="Search code, description..."
   212→   212→                value={search}
   213→   213→                onChange={(e) => setSearch(e.target.value)}
   214→   214→                className="border-slate-700 bg-slate-800 pl-9 text-slate-200 placeholder:text-slate-600"
   215→   215→              />
   216→   216→            </div>
   217→   217→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   218→   218→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300">
   219→   219→                <SelectValue placeholder="All Statuses" />
   220→   220→              </SelectTrigger>
   221→   221→              <SelectContent className="border-slate-700 bg-slate-800">
   222→   222→                {['IN_TRANSIT', 'RECEIVED', 'IN_YARD', 'IN_WAREHOUSE', 'DISPATCHED', 'DELIVERED'].map((s) => (
   223→   223→                  <SelectItem key={s} value={s} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">
   224→   224→                    {s.replace(/_/g, ' ')}
   225→   225→                  </SelectItem>
   226→   226→                ))}
   227→   227→              </SelectContent>
   228→   228→            </Select>
   229→   229→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   230→   230→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300">
   231→   231→                <SelectValue placeholder="All Categories" />
   232→   232→              </SelectTrigger>
   233→   233→              <SelectContent className="border-slate-700 bg-slate-800">
   234→   234→                {['HEAVY_LIFT', 'OVERSIZE', 'STANDARD', 'PROJECT_CARGO'].map((c) => (
   235→   235→                  <SelectItem key={c} value={c} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">
   236→   236→                    {c.replace(/_/g, ' ')}
   237→   237→                  </SelectItem>
   238→   238→                ))}
   239→   239→              </SelectContent>
   240→   240→            </Select>
   241→   241→            <Select value={commodityFilter} onValueChange={setCommodityFilter}>
   242→   242→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300">
   243→   243→                <SelectValue placeholder="All Commodities" />
   244→   244→              </SelectTrigger>
   245→   245→              <SelectContent className="border-slate-700 bg-slate-800">
   246→   246→                {['GENERAL', 'MACHINERY', 'STEEL', 'EQUIPMENT', 'MODULE'].map((c) => (
   247→   247→                  <SelectItem key={c} value={c} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">
   248→   248→                    {c}
   249→   249→                  </SelectItem>
   250→   250→                ))}
   251→   251→              </SelectContent>
   252→   252→            </Select>
   253→   253→          </div>
   254→   254→        </CardContent>
   255→   255→      </Card>
   256→   256→
   257→   257→      {/* Table */}
   258→   258→      <Card className="border-slate-800 bg-slate-900/50">
   259→   259→        <CardContent className="p-0">
   260→   260→          <div className="max-h-96 overflow-y-auto">
   261→   261→            <Table>
   262→   262→              <TableHeader>
   263→   263→                <TableRow className="border-slate-800 hover:bg-transparent">
   264→   264→                  <TableHead className="text-xs text-slate-500">Code</TableHead>
   265→   265→                  <TableHead className="text-xs text-slate-500">Description</TableHead>
   266→   266→                  <TableHead className="text-xs text-slate-500 hidden md:table-cell">Weight (kg)</TableHead>
   267→   267→                  <TableHead className="text-xs text-slate-500 hidden lg:table-cell">Dims (L×W×H)</TableHead>
   268→   268→                  <TableHead className="text-xs text-slate-500 hidden sm:table-cell">Category</TableHead>
   269→   269→                  <TableHead className="text-xs text-slate-500">Status</TableHead>
   270→   270→                  <TableHead className="text-xs text-slate-500 hidden lg:table-cell">Location</TableHead>
   271→   271→                  <TableHead className="text-xs text-slate-500 hidden xl:table-cell">Project</TableHead>
   272→   272→                  <TableHead className="text-xs text-slate-500 text-right">Actions</TableHead>
   273→   273→                </TableRow>
   274→   274→              </TableHeader>
   275→   275→              <TableBody>
   276→   276→                {loading
   277→   277→                  ? Array.from({ length: 5 }).map((_, i) => (
   278→   278→                      <TableRow key={i} className="border-slate-800 hover:bg-transparent">
   279→   279→                        {Array.from({ length: 9 }).map((_, j) => (
   280→   280→                          <TableCell key={j} className="py-3"><Skeleton className="h-4 w-16 bg-slate-800" /></TableCell>
   281→   281→                        ))}
   282→   282→                      </TableRow>
   283→   283→                    ))
   284→   284→                  : cargo.length === 0
   285→   285→                    ? (
   286→   286→                      <TableRow className="border-slate-800 hover:bg-transparent">
   287→   287→                        <TableCell colSpan={9} className="text-center py-8 text-slate-500">
   288→   288→                          No cargo items found
   289→   289→                        </TableCell>
   290→   290→                      </TableRow>
   291→   291→                    )
   292→   292→                    : cargo.map((item) => (
   293→   293→                      <TableRow key={item.id} className="border-slate-800 hover:bg-slate-800/50">
   294→   294→                        <TableCell className="py-3 text-xs font-mono font-medium text-amber-400/80">{item.cargoCode}</TableCell>
   295→   295→                        <TableCell className="py-3 text-xs text-slate-300 max-w-[200px] truncate">{item.description}</TableCell>
   296→   296→                        <TableCell className="py-3 text-xs text-slate-400 hidden md:table-cell">{item.weight.toLocaleString()}</TableCell>
   297→   297→                        <TableCell className="py-3 text-xs text-slate-400 hidden lg:table-cell font-mono">
   298→   298→                          {item.length}×{item.width}×{item.height}
   299→   299→                        </TableCell>
   300→   300→                        <TableCell className="py-3 hidden sm:table-cell">
   301→   301→                          <Badge variant="outline" className={`text-[10px] ${categoryStyles[item.liftCategory]}`}>{item.liftCategory.replace(/_/g, ' ')}</Badge>
   302→   302→                        </TableCell>
   303→   303→                        <TableCell className="py-3">
   304→   304→                          <Badge variant="outline" className={`text-[10px] ${statusStyles[item.status]}`}>{item.status.replace(/_/g, ' ')}</Badge>
   305→   305→                        </TableCell>
   306→   306→                        <TableCell className="py-3 text-xs text-slate-400 hidden lg:table-cell">{item.location?.code || '—'}</TableCell>
   307→   307→                        <TableCell className="py-3 text-xs text-slate-400 hidden xl:table-cell max-w-[120px] truncate">{item.project?.name || '—'}</TableCell>
   308→   308→                        <TableCell className="py-3 text-right">
   309→   309→                          <div className="flex items-center justify-end gap-1">
   310→   310→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-cyan-400" onClick={() => setViewing(item)}>
   311→   311→                              <Eye className="h-3.5 w-3.5" />
   312→   312→                            </Button>
   313→   313→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-amber-400" onClick={() => openEdit(item)}>
   314→   314→                              <Pencil className="h-3.5 w-3.5" />
   315→   315→                            </Button>
   316→   316→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-red-400" onClick={() => setDeleting(item)}>
   317→   317→                              <Trash2 className="h-3.5 w-3.5" />
   318→   318→                            </Button>
   319→   319→                          </div>
   320→   320→                        </TableCell>
   321→   321→                      </TableRow>
   322→   322→                    ))}
   323→   323→              </TableBody>
   324→   324→            </Table>
   325→   325→          </div>
   326→   326→        </CardContent>
   327→   327→      </Card>
   328→   328→
   329→   329→      {/* Pagination */}
   330→   330→      {totalPages > 1 && (
   331→   331→        <div className="flex items-center justify-center gap-2">
   332→   332→          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(page - 1)}
   333→   333→            className="border-slate-700 bg-slate-800 text-slate-300 hover:bg-slate-700">
   334→   334→            Previous
   335→   335→          </Button>
   336→   336→          <span className="text-xs text-slate-500">Page {page} of {totalPages}</span>
   337→   337→          <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage(page + 1)}
   338→   338→            className="border-slate-700 bg-slate-800 text-slate-300 hover:bg-slate-700">
   339→   339→            Next
   340→   340→          </Button>
   341→   341→        </div>
   342→   342→      )}
   343→   343→
   344→   344→      {/* Add/Edit Dialog */}
   345→   345→      <Dialog open={showAdd || !!editing} onOpenChange={(open) => { if (!open) { setShowAdd(false); setEditing(null); setForm(emptyForm); } }}>
   346→   346→        <DialogContent className="border-slate-700 bg-slate-900 max-h-[90vh] overflow-y-auto max-w-2xl">
   347→   347→          <DialogHeader>
   348→   348→            <DialogTitle className="text-slate-100">{editing ? 'Edit Cargo' : 'Add New Cargo'}</DialogTitle>
   349→   349→          </DialogHeader>
   350→   350→          <div className="grid gap-4 py-4 sm:grid-cols-2">
   351→   351→            <div className="sm:col-span-2">
   352→   352→              <Label className="text-slate-400">Description *</Label>
   353→   353→              <Input value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })}
   354→   354→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" />
   355→   355→            </div>
   356→   356→            <div>
   357→   357→              <Label className="text-slate-400">Weight (kg) *</Label>
   358→   358→              <Input type="number" value={form.weight} onChange={(e) => setForm({ ...form, weight: e.target.value })}
   359→   359→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" />
   360→   360→            </div>
   361→   361→            <div>
   362→   362→              <Label className="text-slate-400">Dimensions (m)</Label>
   363→   363→              <div className="grid grid-cols-3 gap-2 mt-1">
   364→   364→                <Input placeholder="L" type="number" value={form.length} onChange={(e) => setForm({ ...form, length: e.target.value })}
   365→   365→                  className="border-slate-700 bg-slate-800 text-slate-200 text-center" />
   366→   366→                <Input placeholder="W" type="number" value={form.width} onChange={(e) => setForm({ ...form, width: e.target.value })}
   367→   367→                  className="border-slate-700 bg-slate-800 text-slate-200 text-center" />
   368→   368→                <Input placeholder="H" type="number" value={form.height} onChange={(e) => setForm({ ...form, height: e.target.value })}
   369→   369→                  className="border-slate-700 bg-slate-800 text-slate-200 text-center" />
   370→   370→              </div>
   371→   371→            </div>
   372→   372→            <div>
   373→   373→              <Label className="text-slate-400">Lift Category *</Label>
   374→   374→              <Select value={form.liftCategory} onValueChange={(v) => setForm({ ...form, liftCategory: v })}>
   375→   375→                <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-200 mt-1"><SelectValue placeholder="Select..." /></SelectTrigger>
   376→   376→                <SelectContent className="border-slate-700 bg-slate-800">
   377→   377→                  {(['HEAVY_LIFT', 'OVERSIZE', 'STANDARD', 'PROJECT_CARGO'] as LiftCategory[]).map((c) => (
   378→   378→                    <SelectItem key={c} value={c} className="text-slate-200 focus:bg-slate-700">{c.replace(/_/g, ' ')}</SelectItem>
   379→   379→                  ))}
   380→   380→                </SelectContent>
