   107→// ─── KPI Card ─────────────────────────────────────────────────────────────
   108→function KpiCard({
   109→  icon: Icon, label, value, suffix, loading, color = 'text-amber-400',
   110→  accentBorder = 'border-l-amber-400', trend, page, onNavigate,
   111→}: {
   112→  icon: React.ElementType; label: string; value: string | number;
   113→  suffix?: string; loading: boolean; color?: string; accentBorder?: string;
   114→  trend?: number; page?: WmsPage; onNavigate?: (page: WmsPage) => void;
   115→}) {
   116→  const isPositive = trend !== undefined && trend >= 0;
   117→  return (
   118→    <Card
   119→      className={[
   120→        'dark:border-slate-800 border-slate-200 dark:bg-slate-900/50 bg-white',
   121→        'cursor-pointer select-none',
   122→        'hover:shadow-lg hover:shadow-black/5 dark:hover:shadow-black/20',
   123→        'hover:-translate-y-0.5 active:translate-y-0',
   124→        'transition-all duration-200 ease-out',
   125→        `border-l-[3px] ${accentBorder}`,
   126→      ].join(' ')}
   127→      onClick={() => page && onNavigate?.(page)}
   128→      role="button"
   129→      tabIndex={0}
   130→      onKeyDown={(e) => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); if (page) onNavigate?.(page); } }}
   131→    >
   132→      <CardContent className="p-4">
   133→        <div className="flex items-start justify-between">
   134→          <div className="space-y-2 min-w-0">
   135→            {loading ? (
   136→              <>
   137→                <Skeleton className="h-3 w-20 dark:bg-slate-800 bg-slate-100" />
   138→                <Skeleton className="h-8 w-16 dark:bg-slate-800 bg-slate-100" />
   139→              </>
   140→            ) : (
   141→              <>
   142→                <p className="text-[11px] font-medium dark:text-slate-500 text-slate-400 uppercase tracking-wider">{label}</p>
   143→                <p className="text-3xl font-extrabold tracking-tight dark:text-slate-100 text-slate-900 leading-none">
   144→                  {typeof value === 'number' ? value.toLocaleString() : value}
   145→                  {suffix && (
   146→                    <span className="text-xs font-semibold dark:text-slate-500 text-slate-400 ml-1">{suffix}</span>
   147→                  )}
   148→                </p>
   149→                {trend !== undefined && (
   150→                  <div className={`flex items-center gap-1 text-[11px] font-medium ${isPositive ? 'text-emerald-500 dark:text-emerald-400' : 'text-red-500 dark:text-red-400'}`}>
   151→                    {isPositive ? <ArrowUp className="h-3 w-3" /> : <ArrowDown className="h-3 w-3" />}
   152→                    <span>{Math.abs(trend)}%</span>
   153→                  </div>
   154→                )}
   155→              </>
   156→            )}
   157→          </div>
   158→          <div className="flex flex-col items-end gap-2">
   159→            <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-xl dark:bg-slate-800 bg-slate-100 ${color}`}>
   160→              <Icon className="h-5 w-5" />
   161→            </div>
   162→            {page && (
   163→              <ChevronRight className="h-4 w-4 dark:text-slate-700 text-slate-300 opacity-0 group-hover/card:opacity-100 transition-opacity duration-200" />
   164→            )}
   165→          </div>
   166→        </div>
   167→      </CardContent>
   168→    </Card>
   169→  );
   170→}

   232→// ─── Cargo Status Donut Chart ─────────────────────────────────────────────
   233→function CargoStatusDonutChart({ data, loading, t, locale }: {
   234→  data: DashboardStats['statusBreakdown']; loading: boolean;
   235→  t: (k: string) => string; locale: string;
   236→}) {
   237→  if (loading) return <Skeleton className="h-[300px] w-full dark:bg-slate-800 bg-slate-100 rounded-xl" />;
   238→  if (!data || data.length === 0) return null;
   239→
   240→  const total = data.reduce((s, d) => s + d.count, 0);
   241→  const chartData = data.map((d) => ({ ...d, fill: statusColors[d.status] || '#64748b' }));
   242→
   243→  const renderLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: {
   244→    cx: number; cy: number; midAngle: number; innerRadius: number; outerRadius: number; percent: number;
   245→  }) => {
   246→    if (percent < 0.08) return null;
   247→    const RADIAN = Math.PI / 180;
   248→    const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   249→    const x = cx + radius * Math.cos(-midAngle * RADIAN);
   250→    const y = cy + radius * Math.sin(-midAngle * RADIAN);
   251→    return (
   252→      <text x={x} y={y} textAnchor="middle" dominantBaseline="central" className="text-[10px] font-bold" fill="#f8fafc">
   253→        {`${(percent * 100).toFixed(0)}%`}
   254→      </text>
   255→    );
   256→  };
   257→
   258→  return (
   259→    <Card className="dark:border-slate-800 border-slate-200 dark:bg-slate-900/50 bg-white">
   260→      <CardHeader className="pb-2">
   261→        <CardTitle className="text-sm font-medium dark:text-slate-300 text-slate-700">{t('dashboard.cargoStatusBreakdown')}</CardTitle>
   262→      </CardHeader>
   263→      <CardContent>
   264→        <ChartContainer config={pieChartConfig} className="h-[260px] w-full">
   265→          <PieChart>
   266→            <ChartTooltip
   267→              content={(
   268→                <ChartTooltipContent
   269→                  nameKey="status"
   270→                  hideLabel
   271→                  formatter={(value: number, _name: string, item: { payload?: { status?: string } }) => [
   272→                    value.toLocaleString(),
   273→                    translateStatus(item.payload?.status || '', locale as 'ar' | 'en'),
   274→                  ]}
   275→                />
   276→              )}
   277→            />
   278→            <Pie
   279→              data={chartData}
   280→              cx="50%"
   281→              cy="50%"
   282→              innerRadius={60}
   283→              outerRadius={90}
   284→              dataKey="count"
   285→              nameKey="status"
   286→              label={renderLabel}
   287→              labelLine={false}
   288→              strokeWidth={2}
   289→              stroke="transparent"
   290→            >
   291→              {chartData.map((entry, index) => (
   292→                <Cell key={`cell-${index}`} fill={entry.fill} />
   293→              ))}
   294→            </Pie>
   295→            <ChartLegend
   296→              content={({ payload }) => (
   297→                <div className="flex flex-wrap items-center justify-center gap-3 pt-2">
   298→                  {payload?.map((item) => (
   299→                    <div key={item.value} className="flex items-center gap-1.5">
   300→                      <div className="h-2.5 w-2.5 rounded-full" style={{ backgroundColor: item.color }} />
   301→                      <span className="text-[11px] dark:text-slate-400 text-slate-600">
   302→                        {translateStatus(item.value as string, locale as 'ar' | 'en')} ({item.payload?.count || 0})
   303→                      </span>
   304→                    </div>
   305→                  ))}
   306→                </div>
   307→              )}
   308→            />
   309→          </PieChart>
   310→        </ChartContainer>
   311→      </CardContent>
   312→    </Card>
   313→  );
   314→}
