   101→   101→// ==================== WEIGHT CATEGORIES ====================
   102→   102→
   103→   103→interface WeightCategory {
   104→   104→  label: string;
   105→   105→  maxTonnes: number;
   106→   106→  color: string;
   107→   107→  colorDark: string;
   108→   108→}
   109→   109→
   110→   110→const WEIGHT_CATEGORIES: WeightCategory[] = [
   111→   111→  { label: 'Light', maxTonnes: 5, color: 'bg-emerald-400', colorDark: 'bg-emerald-500' },
   112→   112→  { label: 'Medium', maxTonnes: 50, color: 'bg-amber-400', colorDark: 'bg-amber-500' },
   113→   113→  { label: 'Heavy', maxTonnes: 200, color: 'bg-orange-500', colorDark: 'bg-orange-500' },
   114→   114→  { label: 'Super Heavy', maxTonnes: 500, color: 'bg-red-500', colorDark: 'bg-red-500' },
   115→   115→];
   116→   116→
   117→   117→function getWeightCategory(weightKg: number): WeightCategory {
   118→   118→  const tonnes = weightKg / 1000;
   119→   119→  for (let i = WEIGHT_CATEGORIES.length - 1; i >= 0; i--) {
   120→   120→    if (tonnes > WEIGHT_CATEGORIES[i].maxTonnes) {
   121→   121→      return { ...WEIGHT_CATEGORIES[i] };
   122→   122→    }
   123→   123→  }
   124→   124→  return WEIGHT_CATEGORIES[0];
   125→   125→}
   126→   126→
   127→   127→function getWeightPercentage(weightKg: number): number {
   128→   128→  const tonnes = weightKg / 1000;
   129→   129→  return Math.min(100, (tonnes / 500) * 100);
   130→   130→}
