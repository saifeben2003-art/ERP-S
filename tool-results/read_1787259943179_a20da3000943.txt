   107→// ─── KPI Card ─────────────────────────────────────────────────────────────
   108→function KpiCard({
   109→  icon: Icon, label, value, suffix, loading, color = 'text-amber-400',
   110→  accentBorder = 'border-l-amber-400', trend, page, onNavigate,
   111→}: {
   112→  icon: React.ElementType; label: string; value: string | number;
   113→  suffix?: string; loading: boolean; color?: string; accentBorder?: string;
   114→  trend?: number; page?: WmsPage; onNavigate?: (page: WmsPage) => void;
   115→}) {
   116→  const isPositive = trend !== undefined && trend >= 0;
   117→  return (
   118→    <Card
   119→      className={[
   120→        'dark:border-slate-800 border-slate-200 dark:bg-slate-900/50 bg-white',
   121→        'cursor-pointer select-none',
   122→        'hover:shadow-lg hover:shadow-black/5 dark:hover:shadow-black/20',
   123→        'hover:-translate-y-0.5 active:translate-y-0',
   124→        'transition-all duration-200 ease-out',
   125→        `border-l-[3px] ${accentBorder}`,
   126→      ].join(' ')}
   127→      onClick={() => page && onNavigate?.(page)}
   128→      role="button"
   129→      tabIndex={0}
   130→      onKeyDown={(e) => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); if (page) onNavigate?.(page); } }}
   131→    >
   132→      <CardContent className="p-4">
   133→        <div className="flex items-start justify-between">
   134→          <div className="space-y-2 min-w-0">
   135→            {loading ? (
   136→              <>
   137→                <Skeleton className="h-3 w-20 dark:bg-slate-800 bg-slate-100" />
   138→                <Skeleton className="h-8 w-16 dark:bg-slate-800 bg-slate-100" />
   139→              </>
   140→            ) : (
   141→              <>
   142→                <p className="text-[11px] font-medium dark:text-slate-500 text-slate-400 uppercase tracking-wider">{label}</p>
   143→                <p className="text-3xl font-extrabold tracking-tight dark:text-slate-100 text-slate-900 leading-none">
   144→                  {typeof value === 'number' ? value.toLocaleString() : value}
   145→                  {suffix && (
   146→                    <span className="text-xs font-semibold dark:text-slate-500 text-slate-400 ml-1">{suffix}</span>
   147→                  )}
   148→                </p>
   149→                {trend !== undefined && (
   150→                  <div className={`flex items-center gap-1 text-[11px] font-medium ${isPositive ? 'text-emerald-500 dark:text-emerald-400' : 'text-red-500 dark:text-red-400'}`}>
   151→                    {isPositive ? <ArrowUp className="h-3 w-3" /> : <ArrowDown className="h-3 w-3" />}
   152→                    <span>{Math.abs(trend)}%</span>
   153→                  </div>
   154→                )}
   155→              </>
   156→            )}
   157→          </div>
   158→          <div className="flex flex-col items-end gap-2">
   159→            <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-xl dark:bg-slate-800 bg-slate-100 ${color}`}>
   160→              <Icon className="h-5 w-5" />
   161→            </div>
   162→            {page && (
   163→              <ChevronRight className="h-4 w-4 dark:text-slate-700 text-slate-300 opacity-0 group-hover/card:opacity-100 transition-opacity duration-200" />
   164→            )}
   165→          </div>
   166→        </div>
   167→      </CardContent>
   168→    </Card>
   169→  );
   170→}

   233→// ─── Cargo Status Donut Chart ─────────────────────────────────────────────
   234→function CargoStatusDonutChart({ data, loading, t, locale }: {
   235→  data: DashboardStats['statusBreakdown']; loading: boolean;
   236→  t: (k: string) => string; locale: string;
   237→}) {
   238→  if (loading) return <Skeleton className="h-[300px] w-full dark:bg-slate-800 bg-slate-100 rounded-xl" />;
   239→  if (!data || data.length === 0) return null;
   240→
   241→  const total = data.reduce((s, d) => s + d.count, 0);
   242→  const chartData = data.map((d) => ({ ...d, fill: statusColors[d.status] || '#64748b' }));
   243→
   244→  const renderLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: {
   245→    cx: number; cy: number; midAngle: number; innerRadius: number; outerRadius: number; percent: number;
   246→  }) => {
   247→    if (percent < 0.08) return null;
   248→    const RADIAN = Math.PI / 180;
   249→    const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   250→    const x = cx + radius * Math.cos(-midAngle * RADIAN);
   251→    const y = cy + radius * Math.sin(-midAngle * RADIAN);
   252→    return (
   253→      <text x={x} y={y} textAnchor="middle" dominantBaseline="central" className="text-[10px] font-bold" fill="#f8fafc">
   254→        {`${(percent * 100).toFixed(0)}%`}
   255→      </text>
   256→    );
   257→  };
   258→
   259→  return (
   260→    <Card className="dark:border-slate-800 border-slate-200 dark:bg-slate-900/50 bg-white">
   261→      <CardHeader className="pb-2">
   262→        <CardTitle className="text-sm font-medium dark:text-slate-300 text-slate-700">{t('dashboard.cargoStatusBreakdown')}</CardTitle>
   263→      </CardHeader>
   264→      <CardContent>
   265→        <ChartContainer config={pieChartConfig} className="h-[260px] w-full">
   266→          <PieChart>
   267→            <ChartTooltip
   268→              content={(
   269→                <ChartTooltipContent
   270→                  nameKey="status"
   271→                  hideLabel
   272→                  // eslint-disable-next-line @typescript-eslint/no-explicit-any
   273→                  formatter={(value: any, _name: any, item: any) => [
   274→                    Number(value).toLocaleString(),
   275→                    translateStatus(String(item?.payload?.status || ''), locale as 'ar' | 'en'),
   276→                  ]}
   277→                />
   278→              )}
   279→            />
   280→            <Pie
   281→              data={chartData}
   282→              cx="50%"
   283→              cy="50%"
   284→              innerRadius={60}
   285→              outerRadius={90}
   286→              dataKey="count"
   287→              nameKey="status"
   288→              label={renderLabel}
   289→              labelLine={false}
   290→              strokeWidth={2}
   291→              stroke="transparent"
   292→            >
   293→              {chartData.map((entry, index) => (
   294→                <Cell key={`cell-${index}`} fill={entry.fill} />
   295→              ))}
   296→            </Pie>
   297→            <ChartLegend
   298→              content={({ payload }) => (
   299→                <div className="flex flex-wrap items-center justify-center gap-3 pt-2">
   300→                  {payload?.map((item) => (
   301→                    <div key={item.value} className="flex items-center gap-1.5">
   302→                      <div className="h-2.5 w-2.5 rounded-full" style={{ backgroundColor: item.color }} />
   303→                      <span className="text-[11px] dark:text-slate-400 text-slate-600">
   304→                        {translateStatus(item.value as string, locale as 'ar' | 'en')} ({(item.payload as any)?.count || 0})
   305→                      </span>
   306→                    </div>
   307→                  ))}
   308→                </div>
   309→              )}
   310→            />
   311→          </PieChart>
   312→        </ChartContainer>
   313→      </CardContent>
   314→    </Card>
   315→  );
   316→}
