    58→};
    59→
    60→const statusTabsKeys = [
    61→  { value: '', key: 'common.all' },
    62→  { value: 'PLANNED', key: 'status.PLANNED' },
    63→  { value: 'RECEIVING', key: 'status.RECEIVING' },
    64→  { value: 'IN_STORAGE', key: 'status.IN_STORAGE' },

    68→  { value: 'COMPLETED', key: 'status.COMPLETED' },
    69→];
    70→

    72→  name: '', description: '', clientName: '', clientContact: '',
    73→  destination: '', shippingLine: '', vesselName: '', etd: '', eta: '',
    74→};
    75→
    76→interface ProjectListResponse { items: Project[]; total: number; }
    77→
    78→export function ProjectsPage() {
    79→  const [projects, setProjects] = useState<Project[]>([]);
    80→  const [loading, setLoading] = useState(true);
    81→  const [statusFilter, setStatusFilter] = useState('');
    82→  const [showAdd, setShowAdd] = useState(false);
    83→  const [selectedProject, setSelectedProject] = useState<Project | null>(null);
    84→  const [projectCargo, setProjectCargo] = useState<CargoItem[]>([]);
    85→  const [projectMovements, setProjectMovements] = useState<Movement[]>([]);
    86→  const [detailLoading, setDetailLoading] = useState(false);
    87→  const [statusChanging, setStatusChanging] = useState(false);
    88→  const [form, setForm] = useState(emptyForm);
    89→  const [submitting, setSubmitting] = useState(false);
    90→  const { t } = useTranslation();
    91→
    92→  const fetchProjects = useCallback(async () => {
    93→    setLoading(true);
    94→    const params = new URLSearchParams({ limit: '100', ...(statusFilter && { status: statusFilter }) });
    95→    try {
    96→      const res = await fetch(`/api/projects?${params}`);
    97→      const data: ProjectListResponse = await res.json();
    98→      if (!res.ok) { setProjects([]); return; }
    99→      setProjects(data.items || []);
   100→    } catch { toast.error(t('projects.toast.fetchFailed')); } finally { setLoading(false); }
   101→  }, [statusFilter, t]);
   102→
   103→  useEffect(() => { fetchProjects(); }, [fetchProjects]);
   104→
   105→  const handleCreate = async () => {
   106→    setSubmitting(true);
   107→    try {
   108→      const res = await fetch('/api/projects', {
   109→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   110→        body: JSON.stringify({ name: form.name, description: form.description || null, clientName: form.clientName, clientContact: form.clientContact || null, destination: form.destination || null, shippingLine: form.shippingLine || null, vesselName: form.vesselName || null, etd: form.etd || null, eta: form.eta || null }),
   111→      });
   112→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   113→      toast.success(t('projects.toast.created')); setShowAdd(false); setForm(emptyForm); fetchProjects();
   114→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : t('projects.toast.createFailed')); } finally { setSubmitting(false); }
   115→  };
   116→
   117→  const openProjectDetail = async (project: Project) => {
   118→    setSelectedProject(project);
   119→    setDetailLoading(true);
   120→    try {
   121→      const [cargoRes, moveRes] = await Promise.all([
   122→        fetch(`/api/cargo?projectId=${project.id}&limit=200`),
   123→        fetch(`/api/movements?limit=100`),
   124→      ]);
   125→      const cargoData = await cargoRes.json();
   126→      const moveData = await moveRes.json();
   127→      const allMoves: Movement[] = moveData.items || [];
   128→      const cargoIds = new Set((cargoData.items || []).map((c: CargoItem) => c.id));
   129→      setProjectCargo(cargoData.items || []);
   130→      setProjectMovements(allMoves.filter((m) => cargoIds.has(m.cargoItemId)));
   131→    } catch { toast.error(t('projects.toast.cargoFetchFailed')); } finally { setDetailLoading(false); }
   132→  };
   133→
   134→  const handleProjectStatusChange = async (newStatus: ProjectStatus) => {
   135→    if (!selectedProject) return;
   136→    setStatusChanging(true);
   137→    try {
   138→      const res = await fetch(`/api/projects/${selectedProject.id}/status`, {
   139→        method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ status: newStatus }),
   140→      });
   141→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Failed'); }
   142→      toast.success(t('detail.project.statusChanged'));
   143→      const data = await res.json();
   144→      setSelectedProject(data.data);
   145→      fetchProjects();
   146→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : t('detail.statusChangeFailed')); } finally { setStatusChanging(false); }
   147→  };
   148→
   149→  const currentStep = selectedProject ? PROJECT_STATUSES.indexOf(selectedProject.status) : -1;
   150→
   151→  const cargoStatusCounts = projectCargo.reduce((acc, c) => { acc[c.status] = (acc[c.status] || 0) + 1; return acc; }, {} as Record<string, number>);
   152→
   153→  return (
   154→    <div className="space-y-6">
   155→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   156→        <div>
   157→          <h1 className="text-2xl font-bold text-slate-100">{t('projects.title')}</h1>
   158→          <p className="text-sm text-slate-500 mt-1">{t('projects.subtitle')}</p>
   159→        </div>
   160→        <Button onClick={() => { setForm(emptyForm); setShowAdd(true); }} className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium">
   161→          <Plus className="h-4 w-4 ml-2" /> {t('projects.addProject')}
   162→        </Button>
   163→      </div>
   164→
   165→      {/* Status Tabs */}
   166→      <Tabs value={statusFilter} onValueChange={setStatusFilter}>
   167→        <TabsList className="bg-slate-800/50 border border-slate-700/50 h-auto p-1 flex-wrap gap-1">
   168→          {statusTabsKeys.map((tab) => (
   169→            <TabsTrigger key={tab.value} value={tab.value} className="data-[state=active]:bg-amber-500/15 data-[state=active]:text-amber-400 text-slate-400 text-xs px-3 py-1.5">{t(tab.key)}</TabsTrigger>
   170→          ))}
   171→        </TabsList>
   172→      </Tabs>
   173→
   174→      {/* ========== PROJECT DETAIL VIEW ========== */}
   175→      {selectedProject ? (
   176→        <div className="space-y-4">
   177→          <Button variant="ghost" onClick={() => setSelectedProject(null)} className="text-slate-400 hover:text-slate-200 -mr-2">
   178→            <ArrowRight className="h-4 w-4 ml-2" /> {t('projects.backToProjects')}
   179→          </Button>
   180→
   181→          {/* Project Header Card */}
   182→          <Card className="border-slate-800 bg-slate-900/50">
   183→            <CardContent className="p-6">
   184→              <div className="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
   185→                <div className="min-w-0">
   186→                  <h2 className="text-lg font-bold text-slate-100">{selectedProject.name}</h2>
   187→                  <p className="text-sm font-mono text-slate-600 mt-0.5">{selectedProject.projectCode}</p>
   188→                  {selectedProject.description && <p className="text-xs text-slate-400 mt-2 max-w-lg">{selectedProject.description}</p>}
   189→                </div>
   190→                <Badge variant="outline" className={`shrink-0 text-xs ${projectStatusStyles[selectedProject.status]}`}>{translateStatus(selectedProject.status)}</Badge>
   191→              </div>
   192→
   193→              {/* Project Status Workflow */}
   194→              <div className="mt-5 rounded-xl border border-slate-800 bg-slate-900/60 p-3">
   195→                <div className="flex items-center justify-between mb-2">
   196→                  <h3 className="text-xs font-semibold text-slate-400">{t('detail.project.statusManagement')}</h3>
   197→                </div>
   198→                <div className="flex items-center gap-1 overflow-x-auto pb-1">
   199→                  {PROJECT_STATUSES.map((s, idx) => {
   200→                    const isCompleted = idx <= currentStep;
   201→                    const isCurrent = idx === currentStep;
   202→                    const isNext = PROJECT_WORKFLOW[selectedProject.status]?.includes(s);
   203→                    return (
   204→                      <div key={s} className="flex items-center shrink-0">
   205→                        <button
   206→                          disabled={!isNext || statusChanging}
   207→                          onClick={() => isNext && handleProjectStatusChange(s)}
   208→                          className={`flex flex-col items-center gap-1 px-2.5 py-1.5 rounded-lg transition-all min-w-[60px] ${
   209→                            isCurrent ? 'bg-amber-500/15 ring-1 ring-amber-500/40' :
   210→                            isCompleted ? 'opacity-50' :
   211→                            isNext ? 'hover:bg-slate-800 cursor-pointer' : 'opacity-25'
   212→                          }`}
   213→                        >
   214→                          <div className={`h-2.5 w-2.5 rounded-full ${isCurrent ? 'bg-amber-400' : isCompleted ? 'bg-emerald-400' : 'bg-slate-700'}`} />
   215→                          <span className={`text-[9px] leading-tight text-center ${isCurrent ? 'text-amber-400 font-semibold' : 'text-slate-500'}`}>{translateStatus(s)}</span>
   216→                        </button>
   217→                        {idx < PROJECT_STATUSES.length - 1 && (
   218→                          <div className={`w-6 h-px mx-0.5 ${idx < currentStep ? 'bg-emerald-400/50' : 'bg-slate-800'}`} />
   219→                        )}
   220→                      </div>
   221→                    );
   222→                  })}
   223→                </div>
   224→              </div>
   225→
   226→              {/* Project Info Grid */}
   227→              <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4 mt-5">
   228→                <div className="rounded-lg border border-slate-800 bg-slate-900/60 p-3">
   229→                  <div className="flex items-center gap-2 text-slate-500 mb-1"><Package className="h-3.5 w-3.5" /><span className="text-[10px] uppercase tracking-wider">{t('detail.project.stats.totalItems')}</span></div>
   230→                  <p className="text-xl font-bold text-slate-100">{selectedProject.totalItems}</p>
   231→                </div>
   232→                <div className="rounded-lg border border-slate-800 bg-slate-900/60 p-3">
   233→                  <div className="flex items-center gap-2 text-slate-500 mb-1"><Weight className="h-3.5 w-3.5" /><span className="text-[10px] uppercase tracking-wider">{t('detail.project.stats.totalWeight')}</span></div>
   234→                  <p className="text-xl font-bold text-slate-100">{selectedProject.totalWeight.toLocaleString()}<span className="text-sm font-normal text-slate-500 mr-1">{t('common.tonnes')}</span></p>
   235→                </div>
   236→                <div className="rounded-lg border border-slate-800 bg-slate-900/60 p-3">
   237→                  <div className="flex items-center gap-2 text-slate-500 mb-1"><Volume2 className="h-3.5 w-3.5" /><span className="text-[10px] uppercase tracking-wider">{t('detail.project.stats.totalVolume')}</span></div>
   238→                  <p className="text-xl font-bold text-slate-100">{selectedProject.totalVolume}<span className="text-sm font-normal text-slate-500 mr-1">m³</span></p>
   239→                </div>
   240→                <div className="rounded-lg border border-slate-800 bg-slate-900/60 p-3">
   241→                  <div className="flex items-center gap-2 text-slate-500 mb-1"><MapPin className="h-3.5 w-3.5" /><span className="text-[10px] uppercase tracking-wider">{t('common.location')}</span></div>
   242→                  <p className="text-sm font-medium text-slate-200 truncate mt-0.5">{selectedProject.destination || t('projects.tbd')}</p>
   243→                </div>
   244→              </div>
   245→
   246→              {/* Shipping Info */}
   247→              <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4 mt-4 text-sm">
   248→                <div className="flex items-center gap-2 text-slate-400"><Users className="h-4 w-4 text-amber-500/60" /><div><p className="text-[10px] text-slate-600">{t('detail.project.overview.client')}</p><p className="text-slate-200">{selectedProject.clientName}</p></div></div>
   249→                <div className="flex items-center gap-2 text-slate-400"><Ship className="h-4 w-4 text-amber-500/60" /><div><p className="text-[10px] text-slate-600">{t('detail.project.overview.vessel')}</p><p className="text-slate-200">{selectedProject.vesselName || t('projects.tbd')}</p></div></div>
   250→                {(selectedProject.etd || selectedProject.eta) && (
   251→                  <div className="flex items-center gap-2 text-slate-400"><Calendar className="h-4 w-4 text-amber-500/60" /><div><p className="text-[10px] text-slate-600">ETD / ETA</p><p className="text-slate-200">{selectedProject.etd ? new Date(selectedProject.etd).toLocaleDateString() : '—'} / {selectedProject.eta ? new Date(selectedProject.eta).toLocaleDateString() : '—'}</p></div></div>
   252→                )}
   253→                {selectedProject.shippingLine && (
   254→                  <div className="flex items-center gap-2 text-slate-400"><Ship className="h-4 w-4 text-amber-500/60" /><div><p className="text-[10px] text-slate-600">{t('detail.project.overview.shippingLine')}</p><p className="text-slate-200">{selectedProject.shippingLine}</p></div></div>
   255→                )}
   256→              </div>
   257→            </CardContent>
   258→          </Card>
   259→
   260→          {/* Cargo Status Breakdown + Progress */}
   261→          <div className="grid gap-4 lg:grid-cols-3">
   262→            <Card className="border-slate-800 bg-slate-900/50 lg:col-span-2">
   263→              <CardContent className="p-4">
   264→                <div className="flex items-center justify-between mb-3">
   265→                  <h3 className="text-sm font-medium text-slate-300">{t('projects.cargoItems')}</h3>
   266→                  <span className="text-xs text-slate-500">{projectCargo.length} {t('common.items')}</span>
   267→                </div>
   268→                {detailLoading ? (
   269→                  <div className="space-y-2">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-10 bg-slate-800" />)}</div>
   270→                ) : projectCargo.length === 0 ? (
   271→                  <p className="text-center text-sm text-slate-500 py-8">{t('projects.noCargoItems')}</p>
   272→                ) : (
   273→                  <div className="max-h-[400px] overflow-y-auto">
   274→                    <Table>
   275→                      <TableHeader>
   276→                        <TableRow className="border-slate-800 hover:bg-transparent">
   277→                          <TableHead className="text-xs text-slate-500">{t('projects.table.code')}</TableHead>
   278→                          <TableHead className="text-xs text-slate-500">{t('cargo.table.description')}</TableHead>
   279→                          <TableHead className="text-xs text-slate-500 hidden sm:table-cell">{t('cargo.table.weight')}</TableHead>
   280→                          <TableHead className="text-xs text-slate-500 hidden md:table-cell">{t('cargo.table.category')}</TableHead>
   281→                          <TableHead className="text-xs text-slate-500">{t('cargo.table.status')}</TableHead>
   282→                          <TableHead className="text-xs text-slate-500 hidden lg:table-cell">{t('cargo.table.location')}</TableHead>
   283→                        </TableRow>
   284→                      </TableHeader>
   285→                      <TableBody>
   286→                        {projectCargo.map((c) => (
   287→                          <TableRow key={c.id} className="border-slate-800 hover:bg-slate-800/50 cursor-pointer">
   288→                            <TableCell className="py-2 text-xs font-mono text-amber-400/80">{c.cargoCode}</TableCell>
   289→                            <TableCell className="py-2 text-xs text-slate-300 max-w-[180px] truncate">{c.description}</TableCell>
   290→                            <TableCell className="py-2 text-xs text-slate-400 hidden sm:table-cell">{c.weight.toLocaleString()}</TableCell>
   291→                            <TableCell className="py-2 hidden md:table-cell"><Badge variant="outline" className="text-[10px] bg-slate-700/50 text-slate-300 border-slate-600">{translateCategory(c.liftCategory)}</Badge></TableCell>
   292→                            <TableCell className="py-2"><Badge variant="outline" className={`text-[10px] ${cargoStatusStyles[c.status] || 'bg-slate-700/50 text-slate-400 border-slate-600'}`}>{translateStatus(c.status)}</Badge></TableCell>
   293→                            <TableCell className="py-2 text-xs text-slate-400 hidden lg:table-cell">{c.location?.code || '—'}</TableCell>
   294→                          </TableRow>
   295→                        ))}
   296→                      </TableBody>
   297→                    </Table>
   298→                  </div>
   299→                )}
   300→              </CardContent>
   301→            </Card>
   302→
   303→            {/* Status Breakdown */}
   304→            <Card className="border-slate-800 bg-slate-900/50">
   305→              <CardContent className="p-4">
   306→                <h3 className="text-sm font-medium text-slate-300 mb-4">{t('detail.project.stats.byStatus')}</h3>
   307→                <div className="space-y-3">
   308→                  {Object.entries(cargoStatusCounts).sort((a, b) => b[1] - a[1]).map(([status, count]) => {
   309→                    const pct = Math.round((count / projectCargo.length) * 100);
   310→                    return (
   311→                      <div key={status} className="space-y-1.5">
   312→                        <div className="flex items-center justify-between">
   313→                          <Badge variant="outline" className={`text-[10px] ${cargoStatusStyles[status] || 'bg-slate-700/50 text-slate-400 border-slate-600'}`}>{translateStatus(status)}</Badge>
   314→                          <span className="text-xs text-slate-400">{count} <span className="text-slate-600">({pct}%)</span></span>
   315→                        </div>
   316→                        <div className="h-1.5 w-full overflow-hidden rounded-full bg-slate-800">
   317→                          <div className="h-full rounded-full bg-gradient-to-r from-amber-500 to-orange-500 transition-all duration-500" style={{ width: `${pct}%` }} />
   318→                        </div>
   319→                      </div>
   320→                    );
   321→                  })}
   322→                </div>
   323→              </CardContent>
   324→            </Card>
   325→          </div>
   326→
   327→          {/* Recent Project Movements */}
   328→          <Card className="border-slate-800 bg-slate-900/50">
   329→            <CardContent className="p-4">
   330→              <h3 className="text-sm font-medium text-slate-300 mb-3">{t('detail.movementHistory')}</h3>
   331→              {projectMovements.length === 0 ? (
   332→                <p className="text-xs text-slate-500 text-center py-4">{t('detail.noMovements')}</p>
   333→              ) : (
   334→                <div className="max-h-60 overflow-y-auto">
   335→                  <Table>
   336→                    <TableHeader>
   337→                      <TableRow className="border-slate-800 hover:bg-transparent">
   338→                        <TableHead className="text-xs text-slate-500">{t('movements.table.ref')}</TableHead>
   339→                        <TableHead className="text-xs text-slate-500">{t('movements.table.cargo')}</TableHead>
   340→                        <TableHead className="text-xs text-slate-500">{t('movements.table.type')}</TableHead>
   341→                        <TableHead className="text-xs text-slate-500 hidden sm:table-cell">{t('movements.table.from')}</TableHead>
   342→                        <TableHead className="text-xs text-slate-500 hidden sm:table-cell">{t('movements.table.to')}</TableHead>
   343→                        <TableHead className="text-xs text-slate-500">{t('common.date')}</TableHead>
   344→                      </TableRow>
   345→                    </TableHeader>
   346→                    <TableBody>
   347→                      {projectMovements.slice(0, 20).map((m) => (
   348→                        <TableRow key={m.id} className="border-slate-800 hover:bg-slate-800/50">
